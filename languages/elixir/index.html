prime_sieve

Source: examples/elixir/prime_sieve.exs

limit = 1_000_000
# atomics is 1-based. 0 = candidate, 1 = composite.
a = :atomics.new(limit, signed: false)
:atomics.put(a, 1, 1)

mark = fn mark, i ->
  if i * i > limit do
    :ok
  else
    if :atomics.get(a, i) == 0 do
      strike = fn strike, j ->
        if j > limit do
          :ok
        else
          :atomics.put(a, j, 1)
          strike.(strike, j + i)
        end
      end
      strike.(strike, i * i)
    end
    mark.(mark, i + 1)
  end
end

mark.(mark, 2)

count =
  Enum.reduce(2..limit, 0, fn i, acc ->
    if :atomics.get(a, i) == 0, do: acc + 1, else: acc
  end)

IO.puts("Primes up to #{limit}: #{count}")

sorting

Source: examples/elixir/sorting.exs

fill = fn fill, n, seed, acc ->
  if n == 0 do
    acc
  else
    seed2 = Bitwise.band(seed * 1664525 + 1013904223, 0xFFFFFFFF)
    fill.(fill, n - 1, seed2, [Bitwise.band(seed2, 0x7FFFFFFF) | acc])
  end
end

n = 1_000_000
data = fill.(fill, n, 42, [])
sorted = Enum.sort(data)
IO.puts("Sorted #{n} integers: first=#{hd(sorted)}, last=#{List.last(sorted)}")
