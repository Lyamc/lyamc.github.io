Source: examples/kotlin/PrimeSieve.kt

fun main() {
    val limit = 1_000_000
    val sieve = BooleanArray(limit + 1) { true }
    sieve[0] = false
    sieve[1] = false
    var i = 2
    while (i * i <= limit) {
        if (sieve[i]) {
            var j = i * i
            while (j <= limit) {
                sieve[j] = false
                j += i
            }
        }
        i++
    }
    var count = 0
    for (k in 2..limit) if (sieve[k]) count++
    println("Primes up to $limit: $count")
}

fun radixSort(a: IntArray) {
    val n = a.size
    val tmp = IntArray(n)
    var src = a
    var dst = tmp
    val count = IntArray(256)
    for (pass in 0 until 4) {
        count.fill(0)
        val shift = pass * 8
        for (i in 0 until n) count[(src[i] ushr shift) and 255]++
        var sum = 0
        for (i in 0 until 256) {
            val c = count[i]
            count[i] = sum
            sum += c
        }
        for (i in 0 until n) {
            val v = src[i]
            val b = (v ushr shift) and 255
            dst[count[b]++] = v
        }
        val t = src
        src = dst
        dst = t
    }
}

fun main() {
    val n = 1_000_000
    val data = IntArray(n)
