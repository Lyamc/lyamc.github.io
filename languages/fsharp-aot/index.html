prime_sieve

Source: examples/fsharp/prime_sieve/Program.fs

open System

[<EntryPoint>]
let main _ =
    let limit = 1_000_000
    let size = (limit + 1) >>> 1
    let sieve = Array.create size 1uy
    sieve.[0] <- 0uy
    let mutable i = 1
    let mutable cont = true
    while cont do
        let p = (i <<< 1) + 1
        if int64 p * int64 p > int64 limit then
            cont <- false
        else
            if sieve.[i] <> 0uy then
                let mutable j = (p * p) >>> 1
                while j < size do
                    sieve.[j] <- 0uy
                    j <- j + p
            i <- i + 1
    let mutable count = 1
    let mutable k = 1
    while k < size do
        if sieve.[k] <> 0uy then count <- count + 1
        k <- k + 1
    Console.WriteLine("Primes up to {0}: {1}", limit, count)
    0

Source: examples/fsharp/sorting/Program.fs

open System

[<EntryPoint>]
let main _ =
    let n = 1_000_000
    let data = Array.zeroCreate n
    let mutable seed = 42u
    for i = 0 to n - 1 do
        seed <- seed * 1664525u + 1013904223u
        data.[i] <- int (seed &&& 0x7fffffffu)
    let tmp = Array.zeroCreate n
    let mutable src, dst = data, tmp
    let count = Array.zeroCreate 256
    for pass = 0 to 3 do
        for i = 0 to 255 do count.[i] <- 0
        let shift = pass * 8
        for i = 0 to n - 1 do
            let b = (src.[i] >>> shift) &&& 255
            count.[b] <- count.[b] + 1
        let mutable sum = 0
        for i = 0 to 255 do
            let c = count.[i]
            count.[i] <- sum
            sum <- sum + c
        for i = 0 to n - 1 do
            let v = src.[i]
            let b = (v >>> shift) &&& 255
            dst.[count.[b]] <- v
            count.[b] <- count.[b] + 1
        let t = src
        src <- dst
        dst <- t
    Console.WriteLine("Sorted {0} integers: first={1}, last={2}", n, data.[0], data.[n - 1])
    0
