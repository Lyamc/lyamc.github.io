prime_sieve

Source: examples/wren/prime_sieve.wren

var limit = 1000000
var sieve = List.filled(limit + 1, true)
sieve[0] = false
sieve[1] = false
var i = 2
while (i * i <= limit) {
  if (sieve[i]) {
    var j = i * i
    while (j <= limit) {
      sieve[j] = false
      j = j + i
    }
  }
  i = i + 1
}
var count = 0
i = 2
while (i <= limit) {
  if (sieve[i]) count = count + 1
  i = i + 1
}
System.print("Primes up to %(limit): %(count)")

sorting

Source: examples/wren/sorting.wren

var n = 1000000
var data = List.filled(n, 0)
var tmp = List.filled(n, 0)
var seed = 42
var a = 1664525
var c = 1013904223
var two32 = 4294967296
var two31 = 2147483648
var i = 0
while (i < n) {
  var x = seed * a + c
  seed = x - two32 * (x / two32).floor
  if (seed >= two31) {
    data[i] = seed - two31
  } else {
    data[i] = seed
  }
  i = i + 1
}

var SHIFT = [1, 256, 65536, 16777216]
var count = List.filled(256, 0)
var src = data
var dst = tmp
var pass = 0
while (pass < 4) {
  var c = 0
  while (c < 256) {
    count[c] = 0
    c = c + 1
  }
  var s = SHIFT[pass]
  i = 0
  while (i < n) {
    var b = ((src[i] / s).floor % 256).truncate
    count[b] = count[b] + 1
    i = i + 1
  }
  var sum = 0
  c = 0
  while (c < 256) {
    var k = count[c]
    count[c] = sum
    sum = sum + k
    c = c + 1
  }
  i = 0
  while (i < n) {
    var v = src[i]
    var b = ((v / s).floor % 256).truncate
    dst[count[b]] = v
    count[b] = count[b] + 1
    i = i + 1
  }
  var t = src
  src = dst
  dst = t
  pass = pass + 1
}

System.print("Sorted %(n) integers: first=%(src[0]), last=%(src[n - 1])")
