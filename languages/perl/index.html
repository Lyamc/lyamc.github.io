prime_sieve

Source: examples/perl/prime_sieve.pl

use strict;
use warnings;
my $limit = 1_000_000;
my @sieve = (1) x ($limit + 1);
$sieve[0] = $sieve[1] = 0;
for (my $i = 2; $i * $i <= $limit; $i++) {
    if ($sieve[$i]) {
        for (my $j = $i * $i; $j <= $limit; $j += $i) {
            $sieve[$j] = 0;
        }
    }
}
my $count = 0;
for my $i (2 .. $limit) { $count++ if $sieve[$i]; }
print "Primes up to $limit: $count\n";

    $seed = ($seed * 1664525 + 1013904223) & 0xffffffff;
    $data[$i] = $seed & 0x7fffffff;
}
@data = sort { $a <=> $b } @data;
