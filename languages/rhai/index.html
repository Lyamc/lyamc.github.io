prime_sieve

Source: examples/rhai/prime_sieve.rhai

let limit = 1_000_000;
let sieve = [];
sieve.pad(limit + 1, true);
sieve[0] = false;
sieve[1] = false;
let i = 2;
while i * i <= limit {
    if sieve[i] {
        let j = i * i;
        while j <= limit {
            sieve[j] = false;
            j += i;
        }
    }
    i += 1;
}
let count = 0;
i = 2;
while i <= limit {
    if sieve[i] { count += 1; }
    i += 1;
}
print(`Primes up to ${limit}: ${count}`);

Source: examples/rhai/sorting.rhai

let n = 1_000_000;
let data = [];
let seed = 42;
for i in 0..n {
    seed = (seed * 1664525 + 1013904223) % 4294967296;
    data.push(seed % 2147483648);
}
data.sort();
print(`Sorted ${n} integers: first=${data[0]}, last=${data[n-1]}`);
