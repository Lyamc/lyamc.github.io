prime_sieve

Source: examples/go/prime_sieve.go

package main

import "fmt"

func main() {
	const limit = 1_000_000
	sieve := make([]bool, limit+1)
	for i := 2; i <= limit; i++ {
		sieve[i] = true
	}
	for i := 2; i*i <= limit; i++ {
		if sieve[i] {
			for j := i * i; j <= limit; j += i {
				sieve[j] = false
			}
		}
	}
	count := 0
	for i := 2; i <= limit; i++ {
		if sieve[i] {
			count++
		}
	}
	fmt.Printf("Primes up to %d: %d\n", limit, count)
}

Source: examples/go/sorting.go

package main

import "fmt"

func radixSort(a []uint32) {
	n := len(a)
	tmp := make([]uint32, n)
	src, dst := a, tmp
	var count [256]uint32
	for pass := 0; pass < 4; pass++ {
		for i := range count {
			count[i] = 0
		}
		shift := uint(pass * 8)
		for _, v := range src {
			count[(v>>shift)&255]++
		}
		var sum uint32
		for i := 0; i < 256; i++ {
			c := count[i]
			count[i] = sum
			sum += c
		}
		for _, v := range src {
			b := (v >> shift) & 255
			dst[count[b]] = v
			count[b]++
		}
		src, dst = dst, src
	}
}

func main() {
	const n = 1_000_000
	data := make([]uint32, n)
	seed := uint32(42)
	for i := 0; i < n; i++ {
		seed = seed*1664525 + 1013904223
		data[i] = seed & 0x7fffffff
	}
	radixSort(data)
	fmt.Printf("Sorted %d integers: first=%d, last=%d\n", n, data[0], data[n-1])
}
