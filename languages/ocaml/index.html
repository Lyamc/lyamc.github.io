Source: examples/ocaml/prime_sieve.ml

let () =
  let limit = 1_000_000 in
  let sieve = Bytes.make (limit + 1) '\001' in
  Bytes.set sieve 0 '\000';
  Bytes.set sieve 1 '\000';
  let rec mark i j =
    if j <= limit then (
      Bytes.set sieve j '\000';
      mark i (j + i))
  in
  let rec loop i =
    if i * i <= limit then (
      if Bytes.get sieve i <> '\000' then mark i (i * i);
      loop (i + 1))
  in
  loop 2;
  let count = ref 0 in
  for i = 2 to limit do
    if Bytes.get sieve i <> '\000' then incr count
  done;
  Printf.printf "Primes up to %d: %d\n" limit !count

let radix_sort (a : int array) =
  let n = Array.length a in
  let tmp = Array.make n 0 in
  let src = ref a in
  let dst = ref tmp in
  let count = Array.make 256 0 in
  for pass = 0 to 3 do
    Array.fill count 0 256 0;
    let shift = pass * 8 in
    let s = !src in
    for i = 0 to n - 1 do
      let b = (s.(i) lsr shift) land 255 in
      count.(b) <- count.(b) + 1
    done;
    let sum = ref 0 in
    for i = 0 to 255 do
      let c = count.(i) in
      count.(i) <- !sum;
      sum := !sum + c
    done;
    let d = !dst in
    for i = 0 to n - 1 do
      let v = s.(i) in
      let b = (v lsr shift) land 255 in
      d.(count.(b)) <- v;
      count.(b) <- count.(b) + 1
    done;
    let t = !src in
    src := !dst;
    dst := t
  done
