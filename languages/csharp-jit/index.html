prime_sieve

Source: examples/csharp/prime_sieve/Program.cs

using System;

class Program
{
    static void Main()
    {
        const int limit = 1_000_000;
        int size = (limit + 1) >> 1;
        byte[] sieve = new byte[size];
        Array.Fill(sieve, (byte)1);
        sieve[0] = 0;
        for (int i = 1; ; i++)
        {
            int p = (i << 1) + 1;
            if ((long)p * p > limit) break;
            if (sieve[i] == 0) continue;
            for (int j = (p * p) >> 1; j < size; j += p)
                sieve[j] = 0;
        }
        int count = 1;
        for (int k = 1; k < size; k++)
            if (sieve[k] != 0) count++;
        Console.WriteLine($"Primes up to {limit}: {count}");
    }
}

Source: examples/csharp/sorting/Program.cs

using System;

class Program
{
    static void RadixSort(uint[] a)
    {
        int n = a.Length;
        uint[] tmp = new uint[n];
        uint[] src = a, dst = tmp;
        Span<int> count = stackalloc int[256];
        for (int pass = 0; pass < 4; pass++)
        {
            count.Clear();
            int shift = pass * 8;
            for (int i = 0; i < n; i++)
                count[(int)((src[i] >> shift) & 255)]++;
            int sum = 0;
            for (int i = 0; i < 256; i++)
            {
                int c = count[i];
                count[i] = sum;
                sum += c;
            }
            for (int i = 0; i < n; i++)
            {
                uint v = src[i];
                int b = (int)((v >> shift) & 255);
                dst[count[b]++] = v;
            }
            (src, dst) = (dst, src);
        }
    }

    static void Main()
    {
        const int N = 1_000_000;
        uint[] data = new uint[N];
        uint seed = 42;
        for (int i = 0; i < N; i++)
        {
            seed = seed * 1664525u + 1013904223u;
            data[i] = seed & 0x7fffffffu;
        }
        RadixSort(data);
        Console.WriteLine($"Sorted {N} integers: first={data[0]}, last={data[N - 1]}");
    }
}
