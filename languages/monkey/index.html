prime_sieve

Source: examples/monkey/prime_sieve.monkey

let limit = 1000000;
let sieve = make_bools(1000001);
set(sieve, 0, 0);
set(sieve, 1, 0);
let i = 2;
while (i * i < 1000001) {
  if (sieve[i]) {
    let j = i * i;
    while (j < 1000001) {
      set(sieve, j, 0);
      let j = j + i;
    }
  }
  let i = i + 1;
}
let count = 0;
let k = 2;
while (k < 1000001) {
  if (sieve[k]) {
    let count = count + 1;
  }
  let k = k + 1;
}
puts("Primes up to 1000000: ");
puts(count);

Source: examples/monkey/sorting.monkey

let n = 1000000;
let data = make_ints(n);
let seed = 42;
let i = 0;
while (i < n) {
  let seed = (seed * 1664525 + 1013904223) % 4294967296;
  set(data, i, seed % 2147483648);
  let i = i + 1;
}
sort(data);
puts("Sorted 1000000 integers: first=");
puts(data[0]);
puts(", last=");
puts(data[n - 1]);
