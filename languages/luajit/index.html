prime_sieve

Source: examples/lua/prime_sieve.lua

local LIMIT = 1000000
local sieve = {}
for i = 2, LIMIT do
  sieve[i] = true
end
for i = 2, math.floor(math.sqrt(LIMIT)) do
  if sieve[i] then
    for j = i * i, LIMIT, i do
      sieve[j] = false
    end
  end
end
local count = 0
for i = 2, LIMIT do
  if sieve[i] then count = count + 1 end
end
print(string.format("Primes up to %d: %d", LIMIT, count))

sorting

Source: examples/lua/sorting.lua

local N = 1000000
local data, tmp = {}, {}
local seed = 42
for i = 1, N do
  seed = (seed * 1664525 + 1013904223) % 4294967296
  data[i] = seed % 2147483648
  tmp[i] = 0
end

-- Digit via / % so the same file parses on Lua 5.4 and LuaJIT 5.1.
local DIV = {1, 256, 65536, 16777216}
local count = {}
local src, dst = data, tmp
for p = 0, 3 do
  for i = 0, 255 do count[i] = 0 end
  local d = DIV[p + 1]
  for i = 1, N do
    local b = math.floor(src[i] / d) % 256
    count[b] = count[b] + 1
  end
  local sum = 0
  for i = 0, 255 do
    local c = count[i]
    count[i] = sum
    sum = sum + c
  end
  for i = 1, N do
    local v = src[i]
    local b = math.floor(v / d) % 256
    count[b] = count[b] + 1
    dst[count[b]] = v
  end
  src, dst = dst, src
end

print(string.format("Sorted %d integers: first=%d, last=%d", N, src[1], src[N]))
