prime_sieve

Source: examples/haskell/prime_sieve.hs

import Data.Array.ST
import Data.Array.Unboxed
import Control.Monad

sieve :: Int -> Int
sieve limit =
  let arr = runSTUArray $ do
        a <- newArray (0, limit) True
        writeArray a 0 False
        writeArray a 1 False
        forM_ [2 .. isqrt limit] $ \i -> do
          p <- readArray a i
          when p $
            forM_ [i * i, i * i + i .. limit] $ \j ->
              writeArray a j False
        return a
  in length [i | i <- [2 .. limit], arr ! i]

isqrt :: Int -> Int
isqrt = floor . sqrt . fromIntegral

main :: IO ()
main = putStrLn $ "Primes up to 1000000: " ++ show (sieve 1000000)

sorting

Source: examples/haskell/sorting.hs

{-# LANGUAGE BangPatterns #-}

import Control.Monad.ST
import Data.Array.Base (unsafeRead, unsafeWrite)
import Data.Array.ST (STUArray, newArray, newArray_)
import Data.Bits ((.&.), unsafeShiftR)
import Data.Word (Word32)

n :: Int
n = 1000000

radix :: STUArray s Int Word32 -> STUArray s Int Word32 -> ST s ()
radix a tmp = go 0 True
  where
    go !pass srcIsA
      | pass == 4 = return ()
      | otherwise = do
          counts <- newArray (0, 255) (0 :: Int) :: ST s (STUArray s Int Int)
          let shift = pass * 8
              src = if srcIsA then a else tmp
              dst = if srcIsA then tmp else a
          let hist !i
                | i >= n = return ()
                | otherwise = do
                    v <- unsafeRead src i
                    let b = fromIntegral ((v `unsafeShiftR` shift) .&. 255)
                    c <- unsafeRead counts b
                    unsafeWrite counts b (c + 1)
                    hist (i + 1)
          hist 0
          let prefix !i !s
                | i >= 256 = return ()
                | otherwise = do
                    c <- unsafeRead counts i
                    unsafeWrite counts i s
                    prefix (i + 1) (s + c)
          prefix 0 0
          let scatter !i
                | i >= n = return ()
                | otherwise = do
                    v <- unsafeRead src i
                    let b = fromIntegral ((v `unsafeShiftR` shift) .&. 255)
                    c <- unsafeRead counts b
                    unsafeWrite dst c v
                    unsafeWrite counts b (c + 1)
                    scatter (i + 1)
          scatter 0
          go (pass + 1) (not srcIsA)

fillLCG :: STUArray s Int Word32 -> ST s ()
fillLCG a = loop 0 (42 :: Word32)
  where
    loop !i !seed
      | i >= n = return ()
      | otherwise = do
          let seed' = seed * 1664525 + 1013904223
          unsafeWrite a i (seed' .&. 0x7fffffff)
          loop (i + 1) seed'

runSort :: (Word32, Word32)
runSort = runST $ do
  a <- newArray_ (0, n - 1)
  tmp <- newArray_ (0, n - 1)
  fillLCG a
  radix a tmp
  first <- unsafeRead a 0
  last_ <- unsafeRead a (n - 1)
  return (first, last_)

main :: IO ()
main = do
  let (first, last_) = runSort
  putStrLn $ "Sorted " ++ show n ++ " integers: first=" ++ show first ++ ", last=" ++ show last_
