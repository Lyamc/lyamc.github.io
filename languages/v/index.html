prime_sieve

Source: examples/v/prime_sieve.v

fn main() {
	limit := 1_000_000
	mut sieve := []bool{len: limit + 1, init: true}
	sieve[0] = false
	sieve[1] = false
	mut i := 2
	for i * i <= limit {
		if sieve[i] {
			mut j := i * i
			for j <= limit {
				sieve[j] = false
				j += i
			}
		}
		i++
	}
	mut count := 0
	for k in 2 .. limit + 1 {
		if sieve[k] {
			count++
		}
	}
	println('Primes up to ${limit}: ${count}')
}

Source: examples/v/sorting.v

fn radix_sort(mut a []u32) {
	n := a.len
	mut tmp := []u32{len: n}
	for pass in 0 .. 4 {
		mut count := [256]u32{}
		shift := u32(pass * 8)
		for i in 0 .. n {
			count[(a[i] >> shift) & 255]++
		}
		mut sum := u32(0)
		for i in 0 .. 256 {
			c := count[i]
			count[i] = sum
			sum += c
		}
		for i in 0 .. n {
			v := a[i]
			b := (v >> shift) & 255
			tmp[count[b]] = v
			count[b]++
		}
		for i in 0 .. n {
			a[i] = tmp[i]
		}
	}
}

fn main() {
	n := 1_000_000
	mut data := []u32{len: n}
	mut seed := u32(42)
	for i in 0 .. n {
		seed = seed * 1664525 + 1013904223
		data[i] = seed & 0x7fffffff
	}
	radix_sort(mut data)
	println('Sorted ${n} integers: first=${data[0]}, last=${data[n - 1]}')
}
