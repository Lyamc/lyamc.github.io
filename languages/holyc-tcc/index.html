I64 Main()
{
  I64 LIMIT = 1000000;
  U8 *sieve = MAlloc(LIMIT + 1);
  I64 i;
  I64 j;
  I64 count;
  if (!sieve)
    return 1;
  for (i = 0; i <= LIMIT; i = i + 1)
    sieve[i] = 1;
  sieve[0] = 0;
  sieve[1] = 0;
  for (i = 2; i * i <= LIMIT; i = i + 1)
  {
    if (sieve[i])
    {
      for (j = i * i; j <= LIMIT; j = j + i)
        sieve[j] = 0;
    }
  }
  count = 0;
  for (i = 2; i <= LIMIT; i = i + 1)
  {
    if (sieve[i])
      count = count + 1;
  }
  $("Primes up to %lld: %lld\n", LIMIT, count);
  Free(sieve);
  return 0;
}

U0 RadixSort(U32 *a, U32 *tmp, I64 n)
{
  U32[256] count;
  U32 *src = a;
  U32 *dst = tmp;
  U32 *t;
  I64 pass;
  I64 shift;
  I64 i;
  U32 sum;
  U32 c;
  U32 v;
  for (pass = 0; pass < 4; pass = pass + 1)
  {
    for (i = 0; i < 256; i = i + 1)
      count[i] = 0;
    shift = pass * 8;
    for (i = 0; i < n; i = i + 1)
      count[(src[i] >> shift) & 255] = count[(src[i] >> shift) & 255] + 1;
    sum = 0;
    for (i = 0; i < 256; i = i + 1)
    {
      c = count[i];
      count[i] = sum;
      sum = sum + c;
    }
    for (i = 0; i < n; i = i + 1)
    {
      v = src[i];
      dst[count[(v >> shift) & 255]] = v;
      count[(v >> shift) & 255] = count[(v >> shift) & 255] + 1;
    }
    t = src;
    src = dst;
    dst = t;
  }
}

I64 Main()
{
  I64 N = 1000000;
  U32 *data = U32 *(MAlloc(N * sizeof(U32)));
  U32 *tmp = U32 *(MAlloc(N * sizeof(U32)));
  U32 seed = 42;
  I64 i;
  if (!data || !tmp)
    return 1;
  for (i = 0; i < N; i = i + 1)
  {
    seed = seed * 1664525 + 1013904223;
    data[i] = seed & 0x7fffffff;
  }
  RadixSort(data, tmp, N);
  $("Sorted %lld integers: first=%lld, last=%lld\n", N, data[0], data[N - 1]);
  Free(U8 *(tmp));
  Free(U8 *(data));
  return 0;
}
