program fibonacci
  implicit none
  integer :: i, n
  integer(kind=8) :: a, b, tmp
  n = 42
  a = 0
  b = 1
  do i = 1, n
    tmp = a + b
    a = b
    b = tmp
  end do
  print *, a
end program fibonacci

Source: examples/fortran/prime_sieve.f90

program prime_sieve
  implicit none
  integer, parameter :: limit = 1000000
  logical, allocatable :: sieve(:)
  integer :: i, j, count
  allocate(sieve(0:limit))
  sieve = .true.
  sieve(0) = .false.
  sieve(1) = .false.
  i = 2
  do while (i * i <= limit)
    if (sieve(i)) then
      j = i * i
      do while (j <= limit)
        sieve(j) = .false.
        j = j + i
      end do
    end if
    i = i + 1
  end do
  count = 0
  do i = 2, limit
    if (sieve(i)) count = count + 1
  end do
  print '(A,I0,A,I0)', 'Primes up to ', limit, ': ', count
  deallocate(sieve)
end program prime_sieve

Source: examples/fortran/sorting.f90

program sorting
  implicit none
  integer, parameter :: n = 1000000
  integer, allocatable :: data(:), tmp(:), cnt(:)
  integer :: i, pass, shift, b, c, sum, v
  integer(kind=8) :: seed
  allocate(data(n), tmp(n), cnt(0:255))
  seed = 42
  do i = 1, n
    seed = iand(seed * 1664525_8 + 1013904223_8, 4294967295_8)
    data(i) = int(iand(seed, 2147483647_8))
  end do
  do pass = 0, 3
    cnt = 0
    shift = pass * 8
    do i = 1, n
      b = iand(ishft(data(i), -shift), 255)
      cnt(b) = cnt(b) + 1
    end do
    sum = 1
    do i = 0, 255
      c = cnt(i)
      cnt(i) = sum
      sum = sum + c
    end do
    do i = 1, n
      v = data(i)
      b = iand(ishft(v, -shift), 255)
      tmp(cnt(b)) = v
      cnt(b) = cnt(b) + 1
    end do
    data = tmp
  end do
  print '(A,I0,A,I0,A,I0)', 'Sorted ', n, ' integers: first=', data(1), ', last=', data(n)
  deallocate(data, tmp, cnt)
end program sorting
