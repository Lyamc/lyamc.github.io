prime_sieve

Source: examples/freebasic/prime_sieve.bas

Const LIMIT As Integer = 1000000
Dim sieve(0 To LIMIT) As Byte
Dim As Integer i, j, count
For i = 0 To LIMIT
    sieve(i) = 1
Next
sieve(0) = 0
sieve(1) = 0
i = 2
While i * i <= LIMIT
    If sieve(i) Then
        j = i * i
        While j <= LIMIT
            sieve(j) = 0
            j += i
        Wend
    End If
    i += 1
Wend
count = 0
For i = 2 To LIMIT
    If sieve(i) Then count += 1
Next
Print "Primes up to "; LIMIT; ": "; count

sorting

Source: examples/freebasic/sorting.bas

' LSD radix sort, 4 passes of 8 bits. ULong is 32-bit on both win32 and win64
' (UInteger is pointer-sized and doubled the traffic on 64-bit).
Const N As Integer = 1000000
Dim Shared vals(0 To N - 1) As ULong
Dim Shared tmp(0 To N - 1) As ULong
Dim Shared hist(0 To 255) As ULong
Dim As ULong seed = 42, sumv, c, v
Dim As Integer i, pass, shift, b
Dim As ULong Ptr src, dst, t, hp, p, e

src = @vals(0)
dst = @tmp(0)
hp = @hist(0)
e = src + N
p = src
While p < e
    seed = seed * 1664525ul + 1013904223ul
    *p = seed And &h7FFFFFFFul
    p += 1
Wend

For pass = 0 To 3
    shift = pass * 8
    For i = 0 To 255
        hp[i] = 0
    Next
    p = src
    e = src + N
    While p < e
        hp[(*p Shr shift) And 255] += 1
        p += 1
    Wend
    sumv = 0
    For i = 0 To 255
        c = hp[i]
        hp[i] = sumv
        sumv += c
    Next
    p = src
    e = src + N
    While p < e
        v = *p
        b = (v Shr shift) And 255
        dst[hp[b]] = v
        hp[b] += 1
        p += 1
    Wend
    t = src
    src = dst
    dst = t
Next

Print "Sorted "; N; " integers: first="; vals(0); ", last="; vals(N - 1)
