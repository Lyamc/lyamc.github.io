prime_sieve

Source: examples/forth/prime_sieve.fs

\ compact sieve using CREATE allot
1000001 constant SZ
create sv SZ allot

: 0sv SZ 0 do 0 sv i + c! loop ;
: 1sv 2 begin dup 1000000 <= while 1 over sv + c! 1+ repeat drop ;

: strike ( p -- )
  dup 2* 
  begin dup 1000000 <= while
    0 over sv + c!
    over +
  repeat 2drop ;

: go
  0sv 1sv
  2 begin
    dup dup * 1000000 <= while
    dup sv + c@ if dup strike then
    1+
  repeat drop ;

: cnt 0 1000001 2 do sv i + c@ if 1+ then loop ;

go
." Primes up to 1000000: " cnt . cr
bye

Source: examples/forth/sorting.fs

\ LSD radix (4 x 8-bit). 32-bit LCG on 64-bit cells. No recursive qsort.
1000000 constant N
variable src
variable dst
variable shiftv
create cnt 256 cells allot

N cells allocate throw src !
N cells allocate throw dst !

: um32 ( u -- u32 ) 4294967295 and ;
: @i ( addr i -- x ) cells + @ ;
: !i ( x addr i -- ) cells + ! ;
: cnt@ ( b -- n ) cells cnt + @ ;
: cnt! ( n b -- ) cells cnt + ! ;

: fill-nums
  42
  N 0 do
    1664525 * 1013904223 + um32
    dup 2147483647 and src @ i !i
  loop drop ;

: zero-cnt  cnt 256 cells erase ;

: prefix-sum
  0
  256 0 do
    i cnt@ over +
    swap i cnt!
  loop drop ;

: pass ( shift -- )
  shiftv !
  zero-cnt
  N 0 do
    src @ i @i shiftv @ rshift 255 and
    dup cnt@ 1+ swap cnt!
  loop
  prefix-sum
  N 0 do
    src @ i @i
    dup shiftv @ rshift 255 and
    dup cnt@
    2dup 1+ swap cnt!
    nip dst @ swap !i
  loop
  src @ dst @ src ! dst ! ;

: radix  0 pass  8 pass  16 pass  24 pass ;
