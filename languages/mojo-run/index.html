prime_sieve

Source: examples/mojo/prime_sieve.mojo

from std.memory.alloc import alloc, dealloc, Layout


def main():
    comptime LIMIT = 1_000_000
    var allocation = alloc(Layout[UInt8](count=LIMIT + 1))
    var sieve = allocation.unsafe_ptr()
    for i in range(LIMIT + 1):
        sieve[unsafe_offset=i] = 1
    sieve[unsafe_offset=0] = 0
    sieve[unsafe_offset=1] = 0
    var i = 2
    while i * i <= LIMIT:
        if sieve[unsafe_offset=i]:
            var j = i * i
            while j <= LIMIT:
                sieve[unsafe_offset=j] = 0
                j += i
        i += 1
    var count: Int = 0
    for k in range(2, LIMIT + 1):
        if sieve[unsafe_offset=k]:
            count += 1
    print(t"Primes up to {LIMIT}: {count}")
    dealloc(allocation^)

sorting

Source: examples/mojo/sorting.mojo

from std.memory.alloc import alloc, dealloc, Layout


def main():
    comptime N = 1_000_000
    var data_a = alloc(Layout[UInt32](count=N))
    var tmp_a = alloc(Layout[UInt32](count=N))
    var src = data_a.unsafe_ptr().as_unsafe_any_origin()
    var dst = tmp_a.unsafe_ptr().as_unsafe_any_origin()
    var seed: UInt32 = 42
    for i in range(N):
        seed = seed * 1664525 + 1013904223
        src[unsafe_offset=i] = seed & 0x7FFFFFFF
    var count = alloc(Layout[UInt32](count=256))
    var cptr = count.unsafe_ptr()
    for pass_i in range(4):
        for b in range(256):
            cptr[unsafe_offset=b] = 0
        var shift = UInt32(pass_i * 8)
        for i in range(N):
            var bucket = Int((src[unsafe_offset=i] >> shift) & 255)
            cptr[unsafe_offset=bucket] += 1
        var sum: UInt32 = 0
        for b in range(256):
            var c = cptr[unsafe_offset=b]
            cptr[unsafe_offset=b] = sum
            sum += c
        for i in range(N):
            var v = src[unsafe_offset=i]
            var bucket = Int((v >> shift) & 255)
            dst[unsafe_offset=Int(cptr[unsafe_offset=bucket])] = v
            cptr[unsafe_offset=bucket] += 1
        var t = src
        src = dst
        dst = t
    print(t"Sorted {N} integers: first={src[unsafe_offset=0]}, last={src[unsafe_offset=N - 1]}")
    dealloc(count^)
    dealloc(tmp_a^)
    dealloc(data_a^)
