prime_sieve

Source: examples/red/prime_sieve.red

Red [Title: "prime_sieve"]
limit: 1000000
; 0 = candidate/prime, 1 = composite. Size must be a reduced integer.
sieve: make vector! reduce ['integer! 32 limit]
sieve/1: 1
i: 2
while [i * i <= limit] [
    if zero? sieve/:i [
        j: i * i
        while [j <= limit] [
            poke sieve j 1
            j: j + i
        ]
    ]
    i: i + 1
]
count: 0
i: 2
while [i <= limit] [
    if zero? sieve/:i [count: count + 1]
    i: i + 1
]
print ["Primes up to" limit ":" count]
