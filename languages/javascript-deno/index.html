prime_sieve

Source: examples/javascript/prime_sieve.js

function main() {
    const LIMIT = 1_000_000;
    const sieve = new Uint8Array(LIMIT + 1);
    sieve[0] = 0;
    sieve[1] = 0;
    for (let i = 2; i * i <= LIMIT; i++) {
        if (sieve[i] === 0) {
            for (let j = i * i; j <= LIMIT; j += i) {
                sieve[j] = 1;
            }
        }
    }
    let count = 0;
    for (let i = 2; i <= LIMIT; i++) {
        if (sieve[i] === 0) count++;
    }
    console.log(`Primes up to ${LIMIT}: ${count}`);
}

main();

function radixSort(a) {
    const n = a.length;
    const tmp = new Int32Array(n);
    let src = a, dst = tmp;
    const count = new Uint32Array(256);
    for (let pass = 0; pass < 4; pass++) {
        count.fill(0);
        const shift = pass * 8;
        for (let i = 0; i < n; i++) count[(src[i] >>> shift) & 255]++;
        let sum = 0;
        for (let i = 0; i < 256; i++) {
            const c = count[i];
            count[i] = sum;
            sum += c;
        }
        for (let i = 0; i < n; i++) {
            const v = src[i];
            const b = (v >>> shift) & 255;
            dst[count[b]++] = v;
        }
        const t = src;
        src = dst;
        dst = t;
    }
}

function main() {
    const n = 1_000_000;
    const data = new Int32Array(n);
    let seed = 42;
    for (let i = 0; i < n; i++) {
        seed = (Math.imul(seed, 1664525) + 1013904223) | 0;
        data[i] = seed & 0x7fffffff;
    }
    radixSort(data);
    console.log(`Sorted ${n} integers: first=${data[0]}, last=${data[n - 1]}`);
}
