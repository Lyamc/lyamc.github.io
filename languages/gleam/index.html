import gleam/int
import gleam/io

fn fib(n: Int, a: Int, b: Int) -> Int {
  case n {
    0 -> a
    _ -> fib(n - 1, b, a + b)
  }
}

pub fn main() {
  io.println(int.to_string(fib(42, 0, 1)))
}

prime_sieve

Source: examples/gleam/prime_sieve/src/prime_sieve.gleam

import gleam/int
import gleam/io
import gleam/string

pub type Atomics

@external(erlang, "atomics", "new")
fn atomics_new(size: Int, options: List(a)) -> Atomics

@external(erlang, "atomics", "put")
fn atomics_put(ref: Atomics, index: Int, value: Int) -> Nil

@external(erlang, "atomics", "get")
fn atomics_get(ref: Atomics, index: Int) -> Int

fn strike(a: Atomics, j: Int, i: Int, limit: Int) -> Nil {
  case j > limit {
    True -> Nil
    False -> {
      atomics_put(a, j, 1)
      strike(a, j + i, i, limit)
    }
  }
}

fn mark(a: Atomics, i: Int, limit: Int) -> Nil {
  case i * i > limit {
    True -> Nil
    False -> {
      case atomics_get(a, i) {
        0 -> strike(a, i * i, i, limit)
        _ -> Nil
      }
      mark(a, i + 1, limit)
    }
  }
}

fn count_primes(a: Atomics, i: Int, limit: Int, acc: Int) -> Int {
  case i > limit {
    True -> acc
    False -> {
      let add = case atomics_get(a, i) {
        0 -> 1
        _ -> 0
      }
      count_primes(a, i + 1, limit, acc + add)
    }
  }
}

pub fn main() {
  let limit = 1_000_000
  let a = atomics_new(limit, [])
  atomics_put(a, 1, 1)
  mark(a, 2, limit)
  let count = count_primes(a, 2, limit, 0)
  io.println(string.concat([
    "Primes up to ",
    int.to_string(limit),
    ": ",
    int.to_string(count),
  ]))
}

sorting

Source: examples/gleam/sorting/src/sorting.gleam

import gleam/int
import gleam/io
import gleam/list
import gleam/string

fn fill(n: Int, seed: Int, acc: List(Int)) -> List(Int) {
  case n {
    0 -> acc
    _ -> {
      let seed2 = int.bitwise_and(seed * 1664525 + 1013904223, 0xffffffff)
      fill(n - 1, seed2, [int.bitwise_and(seed2, 0x7fffffff), ..acc])
    }
  }
}

pub fn main() {
  let n = 1_000_000
  let data = fill(n, 42, [])
  let sorted = list.sort(data, int.compare)
  let assert Ok(first) = list.first(sorted)
  let assert Ok(last) = list.last(sorted)
  io.println(string.concat([
    "Sorted ",
    int.to_string(n),
    " integers: first=",
    int.to_string(first),
    ", last=",
    int.to_string(last),
  ]))
}
