Source: examples/erlang/prime_sieve.erl

#!/usr/bin/env escript
%%! -noshell
-module(prime_sieve).
-export([main/1]).

%% atomics is 1-based. 0 = candidate, 1 = composite. Skip a million ETS inserts.
main(_) ->
    Limit = 1000000,
    A = atomics:new(Limit, [{signed, false}]),
    atomics:put(A, 1, 1),
    mark(A, 2, Limit),
    Count = count(A, 2, Limit, 0),
    io:format("Primes up to ~p: ~p~n", [Limit, Count]).

mark(_A, I, Limit) when I * I > Limit -> ok;
mark(A, I, Limit) ->
    case atomics:get(A, I) of
        0 -> strike(A, I * I, I, Limit);
        _ -> ok
    end,
    mark(A, I + 1, Limit).

strike(_A, J, _I, Limit) when J > Limit -> ok;
strike(A, J, I, Limit) ->
    atomics:put(A, J, 1),
    strike(A, J + I, I, Limit).

count(_A, I, Limit, Acc) when I > Limit -> Acc;
count(A, I, Limit, Acc) ->
    N = case atomics:get(A, I) of 0 -> Acc + 1; _ -> Acc end,
    count(A, I + 1, Limit, N).

#!/usr/bin/env escript
%%! -noshell
-module(sorting).
-export([main/1]).
main(_) ->
    N = 1000000,
    Data = fill(N, 42, []),
    Sorted = lists:sort(Data),
    io:format("Sorted ~p integers: first=~p, last=~p~n",
              [N, hd(Sorted), lists:last(Sorted)]).

fill(0, _, Acc) -> Acc;
fill(I, Seed, Acc) ->
    Seed2 = (Seed * 1664525 + 1013904223) band 16#ffffffff,
    fill(I - 1, Seed2, [Seed2 band 16#7fffffff | Acc]).
