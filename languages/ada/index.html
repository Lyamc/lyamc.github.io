Source: examples/ada/fibonacci.adb

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO;
procedure Fibonacci is
   A, B, Tmp : Long_Long_Integer;
begin
   A := 0; B := 1;
   for I in 1 .. 42 loop
      Tmp := A + B;
      A := B;
      B := Tmp;
   end loop;
   Ada.Integer_Text_IO.Put (Integer (A));
   New_Line;
end Fibonacci;

prime_sieve

Source: examples/ada/prime_sieve.adb

with Ada.Text_IO; use Ada.Text_IO;
procedure Prime_Sieve is
   Limit : constant Integer := 1_000_000;
   type Bool_Arr is array (0 .. Limit) of Boolean;
   Sieve : Bool_Arr := (others => True);
   Count : Integer := 0;
   I, J : Integer;
begin
   Sieve (0) := False;
   Sieve (1) := False;
   I := 2;
   while I * I <= Limit loop
      if Sieve (I) then
         J := I * I;
         while J <= Limit loop
            Sieve (J) := False;
            J := J + I;
         end loop;
      end if;
      I := I + 1;
   end loop;
   for K in 2 .. Limit loop
      if Sieve (K) then
         Count := Count + 1;
      end if;
   end loop;
   Put_Line ("Primes up to 1000000: " & Integer'Image (Count));
end Prime_Sieve;

sorting

Source: examples/ada/sorting.adb

with Ada.Text_IO; use Ada.Text_IO;
with Interfaces; use Interfaces;
procedure Sorting is
   N : constant Integer := 1_000_000;
   type Arr is array (0 .. N - 1) of Unsigned_32;
   Data, Tmp : Arr;
   Count : array (0 .. 255) of Unsigned_32;
   Seed : Unsigned_32 := 42;
   Sum, C, V : Unsigned_32;
   B, Shift : Integer;
begin
   for I in Data'Range loop
      Seed := Seed * 1664525 + 1013904223;
      Data (I) := Seed and 16#7FFFFFFF#;
   end loop;
   for Pass in 0 .. 3 loop
      Count := (others => 0);
      Shift := Pass * 8;
      for I in Data'Range loop
         B := Integer (Shift_Right (Data (I), Shift) and 255);
         Count (B) := Count (B) + 1;
      end loop;
      Sum := 0;
      for I in Count'Range loop
         C := Count (I);
         Count (I) := Sum;
         Sum := Sum + C;
      end loop;
      for I in Data'Range loop
         V := Data (I);
         B := Integer (Shift_Right (V, Shift) and 255);
         Tmp (Integer (Count (B))) := V;
         Count (B) := Count (B) + 1;
      end loop;
      Data := Tmp;
   end loop;
   Put_Line ("Sorted 1000000 integers: first=" & Unsigned_32'Image (Data (0))
             & ", last=" & Unsigned_32'Image (Data (N - 1)));
end Sorting;
