prime_sieve

Source: examples/rust/prime_sieve.rs

fn main() {
    const LIMIT: usize = 1_000_000;
    let mut sieve = vec![1u8; LIMIT + 1];
    sieve[0] = 0;
    sieve[1] = 0;
    let mut i = 2usize;
    while i * i <= LIMIT {
        if sieve[i] != 0 {
            let mut j = i * i;
            while j <= LIMIT {
                sieve[j] = 0;
                j += i;
            }
        }
        i += 1;
    }
    let count = sieve.iter().skip(2).filter(|&&b| b != 0).count();
    println!("Primes up to {}: {}", LIMIT, count);
}

fn radix_sort(a: &mut [u32]) {
    let n = a.len();
    let mut tmp = vec![0u32; n];
    let mut src = 0u8;
    for pass in 0..4 {
        let (from, to) = if src == 0 {
            (&a[..], &mut tmp[..])
        } else {
            (&tmp[..], &mut a[..])
        };
        let mut count = [0u32; 256];
        let shift = pass * 8;
        for &v in from.iter() {
            count[((v >> shift) & 255) as usize] += 1;
        }
        let mut sum = 0u32;
        for c in count.iter_mut() {
            let k = *c;
            *c = sum;
            sum += k;
        }
        for &v in from.iter() {
            let b = ((v >> shift) & 255) as usize;
            to[count[b] as usize] = v;
            count[b] += 1;
        }
        src ^= 1;
    }
}

fn main() {
    const N: usize = 1_000_000;
    let mut data = vec![0u32; N];
    let mut seed: u32 = 42;
    for x in &mut data {
        seed = seed.wrapping_mul(1664525).wrapping_add(1013904223);
        *x = seed & 0x7fffffff;
    }
    radix_sort(&mut data);
    println!("Sorted {N} integers: first={}, last={}", data[0], data[N - 1]);
}
