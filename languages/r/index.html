prime_sieve

Source: examples/r/prime_sieve.R

limit <- 1000000L
sieve <- rep(TRUE, limit)
sieve[1L] <- FALSE
lim <- as.integer(sqrt(limit))
for (i in 2:lim) {
  if (sieve[i]) {
    sieve[seq.int(i * i, limit, i)] <- FALSE
  }
}
count <- sum(sieve)
cat(sprintf("Primes up to %d: %d\n", limit, count))

Source: examples/r/sorting.R

n <- 1000000L
data <- integer(n)
seed <- 42
a <- 1664525
c <- 1013904223
m32 <- 4294967296
m31 <- 2147483648
for (i in seq_len(n)) {
  seed <- (seed * a + c) %% m32
  data[i] <- as.integer(seed %% m31)
}
data <- sort(data)
cat(sprintf("Sorted %d integers: first=%d, last=%d\n", n, data[1], data[n]))
