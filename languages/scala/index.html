prime_sieve

Source: examples/scala/prime_sieve.scala

@main def main() =
  val limit = 1000000
  val sieve = Array.fill(limit + 1)(true)
  sieve(0) = false
  sieve(1) = false
  var i = 2
  while i * i <= limit do
    if sieve(i) then
      var j = i * i
      while j <= limit do
        sieve(j) = false
        j += i
    i += 1
  var count = 0
  i = 2
  while i <= limit do
    if sieve(i) then count += 1
    i += 1
  println(s"Primes up to $limit: $count")

sorting

Source: examples/scala/sorting.scala

@main def main() =
  val n = 1000000
  val data = new Array[Int](n)
  var seed = 42
  var i = 0
  while i < n do
    seed = seed * 1664525 + 1013904223
    data(i) = seed & 0x7fffffff
    i += 1
  radixSort(data)
  println(s"Sorted $n integers: first=${data(0)}, last=${data(n - 1)}")

def radixSort(a: Array[Int]): Unit =
  val n = a.length
  val tmp = new Array[Int](n)
  var src = a
  var dst = tmp
  val count = new Array[Int](256)
  var pass = 0
  while pass < 4 do
    java.util.Arrays.fill(count, 0)
    val shift = pass * 8
    var i = 0
    while i < n do
      count((src(i) >>> shift) & 255) += 1
      i += 1
    var sum = 0
    i = 0
    while i < 256 do
      val c = count(i)
      count(i) = sum
      sum += c
      i += 1
    i = 0
    while i < n do
      val v = src(i)
      val b = (v >>> shift) & 255
      dst(count(b)) = v
      count(b) += 1
      i += 1
    val t = src
    src = dst
    dst = t
    pass += 1
