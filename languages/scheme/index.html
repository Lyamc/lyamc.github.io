prime_sieve

Source: examples/scheme/prime_sieve.scm

(import (chicken base)
        (chicken fixnum)
        srfi-4)

(declare (block)
         (fixnum-arithmetic)
         (usual-integrations)
         (disable-interrupts))

(define limit 1000000)
(define sieve (make-u8vector (fx+ limit 1) 1))
(u8vector-set! sieve 0 0)
(u8vector-set! sieve 1 0)
(let loop ((i 2))
  (when (fx<= (fx* i i) limit)
    (when (fx= (u8vector-ref sieve i) 1)
      (let loop2 ((j (fx* i i)))
        (when (fx<= j limit)
          (u8vector-set! sieve j 0)
          (loop2 (fx+ j i)))))
    (loop (fx+ i 1))))
(define count
  (let loop ((i 2) (c 0))
    (if (fx> i limit)
        c
        (loop (fx+ i 1)
              (if (fx= (u8vector-ref sieve i) 1) (fx+ c 1) c)))))
(display "Primes up to ")
(display limit)
(display ": ")
(display count)
(newline)

sorting

Source: examples/scheme/sorting.scm

(import (chicken base)
        (chicken fixnum)
        srfi-4)

(declare (block)
         (fixnum-arithmetic)
         (usual-integrations)
         (disable-interrupts))

(define n 1000000)
(define mask32 (fx- (fxshl 1 32) 1))
(define mask31 (fx- (fxshl 1 31) 1))
(define data (make-u32vector n 0))
(define tmp (make-u32vector n 0))
(define count (make-u32vector 256 0))

(let loop ((i 0) (seed 42))
  (when (fx< i n)
    (let ((s (fxand (fx+ (fx* seed 1664525) 1013904223) mask32)))
      (u32vector-set! data i (fxand s mask31))
      (loop (fx+ i 1) s))))

(let loop ((pass 0) (src data) (dst tmp))
  (when (fx< pass 4)
    (let ((shift (fx* pass 8)))
      (do ((i 0 (fx+ i 1)))
          ((fx= i 256))
        (u32vector-set! count i 0))
      (do ((i 0 (fx+ i 1)))
          ((fx= i n))
        (let ((b (fxand (fxshr (u32vector-ref src i) shift) 255)))
          (u32vector-set! count b (fx+ (u32vector-ref count b) 1))))
      (let loop-sum ((i 0) (sum 0))
        (when (fx< i 256)
          (let ((c (u32vector-ref count i)))
            (u32vector-set! count i sum)
            (loop-sum (fx+ i 1) (fx+ sum c)))))
      (do ((i 0 (fx+ i 1)))
          ((fx= i n))
        (let* ((v (u32vector-ref src i))
               (b (fxand (fxshr v shift) 255))
               (p (u32vector-ref count b)))
          (u32vector-set! dst p v)
          (u32vector-set! count b (fx+ p 1)))))
    (loop (fx+ pass 1) dst src)))

(display "Sorted ")
(display n)
(display " integers: first=")
(display (u32vector-ref data 0))
(display ", last=")
(display (u32vector-ref data (fx- n 1)))
(newline)
