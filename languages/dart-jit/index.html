import 'dart:typed_data';

void main() {
  const limit = 1000000;
  final sieve = Uint8List(limit + 1);
  sieve.fillRange(2, limit + 1, 1);
  for (var i = 2; i * i <= limit; i++) {
    if (sieve[i] != 0) {
      for (var j = i * i; j <= limit; j += i) {
        sieve[j] = 0;
      }
    }
  }
  var count = 0;
  for (var i = 2; i <= limit; i++) {
    if (sieve[i] != 0) count++;
  }
  print('Primes up to $limit: $count');
}

import 'dart:typed_data';

void radixSort(Uint32List a) {
  final n = a.length;
  final tmp = Uint32List(n);
  var src = a;
  var dst = tmp;
  final count = Uint32List(256);
  for (var pass = 0; pass < 4; pass++) {
    count.fillRange(0, 256, 0);
    final shift = pass * 8;
    for (var i = 0; i < n; i++) {
      count[(src[i] >> shift) & 255]++;
    }
    var sum = 0;
    for (var i = 0; i < 256; i++) {
      final c = count[i];
      count[i] = sum;
      sum += c;
    }
    for (var i = 0; i < n; i++) {
      final v = src[i];
      final b = (v >> shift) & 255;
      dst[count[b]++] = v;
    }
    final t = src;
    src = dst;
    dst = t;
  }
}

void main() {
  const n = 1000000;
  final data = Uint32List(n);
  var seed = 42;
  for (var i = 0; i < n; i++) {
    seed = (seed * 1664525 + 1013904223) & 0xffffffff;
    data[i] = seed & 0x7fffffff;
  }
  radixSort(data);
  print('Sorted $n integers: first=${data.first}, last=${data.last}');
}
