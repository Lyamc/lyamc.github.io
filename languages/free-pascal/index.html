program Fibonacci;
var
  i: integer;
  a, b, tmp: int64;
begin
  a := 0; b := 1;
  for i := 1 to 42 do
  begin
    tmp := a + b;
    a := b;
    b := tmp;
  end;
  writeln(a);
end.

program PrimeSieve;
const
  Limit = 1000000;
var
  sieve: array[0..Limit] of boolean;
  i, j, count: longint;
begin
  for i := 0 to Limit do sieve[i] := true;
  sieve[0] := false; sieve[1] := false;
  i := 2;
  while i * i <= Limit do
  begin
    if sieve[i] then
    begin
      j := i * i;
      while j <= Limit do
      begin
        sieve[j] := false;
        j := j + i;
      end;
    end;
    i := i + 1;
  end;
  count := 0;
  for i := 2 to Limit do if sieve[i] then inc(count);
  writeln('Primes up to ', Limit, ': ', count);
end.

Source: examples/pascal/sorting.pas

program Sorting;
const
  N = 1000000;
var
  data, tmp: array of longword;
  count: array[0..255] of longword;
  i, pass, shift: longint;
  seed, sum, c, v, b: longword;
  srcIsData: boolean;

begin
  setlength(data, N);
  setlength(tmp, N);
  seed := 42;
  for i := 0 to N - 1 do
  begin
    seed := longword(qword(seed) * 1664525 + 1013904223);
    data[i] := seed and $7fffffff;
  end;
  srcIsData := true;
  for pass := 0 to 3 do
  begin
    fillchar(count, sizeof(count), 0);
    shift := pass * 8;
    if srcIsData then
      for i := 0 to N - 1 do
        inc(count[(data[i] shr shift) and 255])
    else
      for i := 0 to N - 1 do
        inc(count[(tmp[i] shr shift) and 255]);
    sum := 0;
    for i := 0 to 255 do
    begin
      c := count[i];
      count[i] := sum;
      inc(sum, c);
    end;
    if srcIsData then
      for i := 0 to N - 1 do
      begin
        v := data[i];
        b := (v shr shift) and 255;
        tmp[count[b]] := v;
        inc(count[b]);
      end
    else
      for i := 0 to N - 1 do
      begin
        v := tmp[i];
        b := (v shr shift) and 255;
        data[count[b]] := v;
        inc(count[b]);
      end;
    srcIsData := not srcIsData;
  end;
  writeln('Sorted ', N, ' integers: first=', data[0], ', last=', data[N - 1]);
end.
