prime_sieve

Source: examples/prolog/prime_sieve.pl

:- initialization(main, main).

% SWI cannot allocate a 1e6-arity term. Store flags in chunks and
% update them in place with nb_setarg/3. Unbound = prime candidate.
chunk_arity(1024).

main :-
    Limit = 1000000,
    sieve_count(Limit, Count),
    format('Primes up to ~w: ~w~n', [Limit, Count]).

sieve_count(Limit, Count) :-
    chunk_arity(A),
    NChunks is Limit // A + 1,
    functor(Chunks, chunks, NChunks),
    init_chunks(1, NChunks, A, Chunks),
    mark_composite(Chunks, A, 0),
    mark_composite(Chunks, A, 1),
    Root is integer(sqrt(Limit)),
    sieve_from(2, Root, Limit, Chunks, A),
    count_from(2, Limit, Chunks, A, 0, Count).

init_chunks(I, N, _, _) :- I > N, !.
init_chunks(I, N, A, Chunks) :-
    functor(C, c, A),
    nb_setarg(I, Chunks, C),
    I1 is I + 1,
    init_chunks(I1, N, A, Chunks).

slot(I, A, ChunkI, ArgI) :-
    ChunkI is I // A + 1,
    ArgI is I mod A + 1.

mark_composite(Chunks, A, I) :-
    slot(I, A, CI, AI),
    arg(CI, Chunks, Chunk),
    nb_setarg(AI, Chunk, 0).

open_slot(Chunks, A, I) :-
    slot(I, A, CI, AI),
    arg(CI, Chunks, Chunk),
    arg(AI, Chunk, V),
    var(V).

sieve_from(I, Root, _, _, _) :- I > Root, !.
sieve_from(I, Root, Limit, Chunks, A) :-
    (   open_slot(Chunks, A, I)
    ->  Start is I * I,
        mark_from(Start, I, Limit, Chunks, A)
    ;   true
    ),
    I1 is I + 1,
    sieve_from(I1, Root, Limit, Chunks, A).

mark_from(J, _, Limit, _, _) :- J > Limit, !.
mark_from(J, Step, Limit, Chunks, A) :-
    mark_composite(Chunks, A, J),
    J1 is J + Step,
    mark_from(J1, Step, Limit, Chunks, A).

count_from(I, Limit, _, _, Acc, Acc) :- I > Limit, !.
count_from(I, Limit, Chunks, A, Acc, Count) :-
    (   open_slot(Chunks, A, I)
    ->  Acc1 is Acc + 1
    ;   Acc1 = Acc
    ),
    I1 is I + 1,
    count_from(I1, Limit, Chunks, A, Acc1, Count).

:- initialization(main, main).
main :-
    N = 1000000,
    fill(N, 42, [], Data),
    msort(Data, Sorted),
    Sorted = [First|_],
    last(Sorted, Last),
    format('Sorted ~w integers: first=~w, last=~w~n', [N, First, Last]).

fill(0, _, Acc, Acc) :- !.
fill(I, Seed, Acc, Out) :-
    Seed2 is (Seed * 1664525 + 1013904223) /\ 4294967295,
    Val is Seed2 /\ 2147483647,
    I1 is I - 1,
    fill(I1, Seed2, [Val|Acc], Out).
