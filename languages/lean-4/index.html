def fib (n : Nat) : Nat :=
  Id.run do
    let mut a := 0
    let mut b := 1
    for _ in [0:n] do
      let t := a + b
      a := b
      b := t
    return a

def main : IO Unit :=
  IO.println (toString (fib 42))

def countPrimes (limit : Nat) : Nat :=
  Id.run do
    let mut s := ByteArray.emptyWithCapacity (limit + 1)
    let mut k := 0
    while k <= limit do
      s := s.push 1
      k := k + 1
    s := s.set! 0 0
    s := s.set! 1 0
    let mut i := 2
    while i * i <= limit do
      if s.get! i != 0 then
        let mut j := i * i
        while j <= limit do
          s := s.set! j 0
          j := j + i
      i := i + 1
    let mut count := 0
    i := 2
    while i <= limit do
      if s.get! i != 0 then
        count := count + 1
      i := i + 1
    return count

def main : IO Unit := do
  let limit := 1000000
  IO.println s!"Primes up to {limit}: {countPrimes limit}"

Source: examples/lean/sorting.lean

/-- Packed little-endian UInt32s in a ByteArray (the unboxed array type). -/
@[inline] def getU32 (a : ByteArray) (i : Nat) : UInt32 :=
  let off := i * 4
  (a.get! off).toUInt32 |||
  ((a.get! (off + 1)).toUInt32 <<< 8) |||
  ((a.get! (off + 2)).toUInt32 <<< 16) |||
  ((a.get! (off + 3)).toUInt32 <<< 24)

@[inline] def setU32 (a : ByteArray) (i : Nat) (v : UInt32) : ByteArray :=
  let off := i * 4
  let a := a.set! off v.toUInt8
  let a := a.set! (off + 1) (v >>> 8).toUInt8
  let a := a.set! (off + 2) (v >>> 16).toUInt8
  a.set! (off + 3) (v >>> 24).toUInt8

def zeros (nbytes : Nat) : ByteArray :=
  Id.run do
    let mut a := ByteArray.emptyWithCapacity nbytes
    let mut i := 0
    while i < nbytes do
      a := a.push 0
      i := i + 1
    return a

def fillLCG (n : Nat) : ByteArray :=
  Id.run do
    let mut a := zeros (n * 4)
    let mut seed : UInt32 := 42
    let mut i := 0
    while i < n do
      seed := seed * 1664525 + 1013904223
      a := setU32 a i (seed &&& (0x7fffffff : UInt32))
      i := i + 1
    return a

def radixSort (src0 : ByteArray) (n : Nat) : ByteArray :=
  Id.run do
    let mut src := src0
    let mut dst := zeros (n * 4)
    let mut pass := 0
    while pass < 4 do
      let shift : UInt32 := UInt32.ofNat (pass * 8)
      let mut counts := Array.replicate 256 (0 : Nat)
      let mut i := 0
      while i < n do
        let b := (getU32 src i >>> shift &&& (255 : UInt32)).toNat
        counts := counts.set! b (counts[b]! + 1)
        i := i + 1
      let mut sum := 0
      let mut b := 0
      while b < 256 do
        let c := counts[b]!
        counts := counts.set! b sum
        sum := sum + c
        b := b + 1
      i := 0
      while i < n do
        let v := getU32 src i
        let bucket := (v >>> shift &&& (255 : UInt32)).toNat
        let c := counts[bucket]!
        dst := setU32 dst c v
        counts := counts.set! bucket (c + 1)
        i := i + 1
      let tmp := src
      src := dst
      dst := tmp
      pass := pass + 1
    return src

def main : IO Unit := do
  let n := 1000000
  let sorted := radixSort (fillLCG n) n
  let first := getU32 sorted 0
  let last := getU32 sorted (n - 1)
  IO.println s!"Sorted {n} integers: first={first}, last={last}"
