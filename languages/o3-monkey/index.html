let fib = fn(n) {
  let a = 0;
  let b = 1;
  let i = 0;
  while (i < n) {
    let t = a + b;
    a = b;
    b = t;
    i = i + 1;
  }
  a
};

puts(fib(42));

prime_sieve

Source: examples/o3-monkey/prime_sieve.monkey

let is_prime = fn(n) {
  if (n < 2) {
    false
  } else {
    let d = 2;
    let result = true;
    while (d * d <= n) {
      if ((n / d) * d == n) {
        result = false;
        d = n;
      } else {
        d = d + 1;
      }
    }
    result
  }
};

let count_primes = fn(limit) {
  let i = 2;
  let acc = 0;
  while (i <= limit) {
    if (is_prime(i)) {
      acc = acc + 1;
    }
    i = i + 1;
  }
  acc
};

puts("Primes up to 400:");
puts(count_primes(400));

let merge = fn(a, b) {
  let out = [];
  let more = true;
  while (more) {
    if (len(a) == 0) {
      more = false
    } else {
      if (len(b) == 0) {
        more = false
      } else {
        if (first(a) < first(b)) {
          out = push(out, first(a));
          a = rest(a);
        } else {
          out = push(out, first(b));
          b = rest(b);
        }
      }
    }
  }
  while (len(a) > 0) {
    out = push(out, first(a));
    a = rest(a);
  }
  while (len(b) > 0) {
    out = push(out, first(b));
    b = rest(b);
  }
  out
};

let take = fn(xs, n) {
  let out = [];
  let i = 0;
  let more = true;
  while (more) {
    if (i >= n) {
      more = false
    } else {
      if (len(xs) == 0) {
        more = false
      } else {
        out = push(out, first(xs));
        xs = rest(xs);
        i = i + 1;
      }
    }
  }
  out
};

let drop = fn(xs, n) {
  let i = 0;
  let more = true;
  while (more) {
    if (i >= n) {
      more = false
    } else {
      if (len(xs) == 0) {
        more = false
      } else {
        xs = rest(xs);
        i = i + 1;
      }
    }
  }
  xs
};

let mergesort = fn(xs) {
  let n = len(xs);
  if (n < 2) {
    xs
  } else {
    let mid = n / 2;
    merge(mergesort(take(xs, mid)), mergesort(drop(xs, mid)))
  }
};

let next_seed = fn(s) {
  let t = s + 17;
  if (t > 10000) {
    t - 10000
  } else {
    t
  }
};

let gen = fn(n, seed) {
  let acc = [];
  let s = seed;
  let i = 0;
  while (i < n) {
    s = next_seed(s);
    acc = push(acc, s);
    i = i + 1;
  }
  acc
};

let data = gen(100, 42);
let sorted = mergesort(data);
puts("Sorted 100 integers: first=");
puts(first(sorted));
puts(", last=");
puts(last(sorted));
