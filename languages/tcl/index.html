set a 0
set b 1

    set a $b
    set b $tmp
}

prime_sieve

Source: examples/tcl/prime_sieve.tcl

set limit 1000000
set sieve [lrepeat [expr {$limit + 1}] 1]
lset sieve 0 0
lset sieve 1 0
for {set i 2} {$i * $i <= $limit} {incr i} {
    if {[lindex $sieve $i]} {
        for {set j [expr {$i * $i}]} {$j <= $limit} {incr j $i} {
            lset sieve $j 0
        }
    }
}
set count 0
for {set i 2} {$i <= $limit} {incr i} {
    if {[lindex $sieve $i]} { incr count }
}
puts "Primes up to $limit: $count"

sorting

Source: examples/tcl/sorting.tcl

set n 1000000
set seed 42
set data {}
for {set i 0} {$i < $n} {incr i} {
    set seed [expr {($seed * 1664525 + 1013904223) & 0xffffffff}]
    lappend data [expr {$seed & 0x7fffffff}]
}
set data [lsort -integer $data]
puts "Sorted $n integers: first=[lindex $data 0], last=[lindex $data end]"
