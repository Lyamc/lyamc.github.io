fibonacci

Source: examples/wgpu/fibonacci.rs

const SHADER: &str = r#"
@group(0) @binding(0) var<storage, read_write> result: array<u32>;

@compute @workgroup_size(1)
fn main() {
    var a: u32 = 0u;
    var b: u32 = 1u;
    for (var i = 0u; i < 42u; i++) {
        let tmp = a + b;
        a = b;
        b = tmp;
    }
    result[0] = a;
}
"#;

fn main() {
    let gpu = wgpu_common::Gpu::new();
    let buf = gpu.storage_empty(4);
    let pipeline = gpu.pipeline(SHADER);
    let bind = gpu.bind(&pipeline, &[&buf]);
    gpu.dispatch(&pipeline, &bind, 1);
    let out = gpu.read_u32s(&buf, 1);
    println!("{}", out[0]);
}

prime_sieve

Source: examples/wgpu/prime_sieve.rs

const SHADER: &str = r#"
struct Params {
    mode: u32,
    p: u32,
    limit: u32,
    _pad: u32,
}

@group(0) @binding(0) var<storage, read_write> sieve: array<u32>;
@group(0) @binding(1) var<storage, read_write> count: array<atomic<u32>>;
@group(0) @binding(2) var<uniform> params: Params;

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid: vec3<u32>) {
    let i = gid.x;
    if params.mode == 0u {
        if i <= params.limit {
            sieve[i] = select(1u, 0u, i < 2u);
        }
    } else if params.mode == 1u {
        let j = params.p * params.p + i * params.p;
        if j <= params.limit {
            sieve[j] = 0u;
        }
    } else if i <= params.limit && sieve[i] == 1u {
        atomicAdd(&count[0], 1u);
    }
}
"#;

fn main() {
    const LIMIT: u32 = 1_000_000;
    let gpu = wgpu_common::Gpu::new();
    let sieve = gpu.storage_empty(((LIMIT + 1) * 4) as u64);
    let count = gpu.storage_init(bytemuck::bytes_of(&0u32));
    let pipeline = gpu.pipeline(SHADER);

    let sqrt = (LIMIT as f64).sqrt() as u32;
    let mut is_prime = vec![true; (sqrt + 1) as usize];
    is_prime[0] = false;
    is_prime[1] = false;
    for p in 2..=sqrt {
        if is_prime[p as usize] {
            let mut j = p * p;
            while j <= sqrt {
                is_prime[j as usize] = false;
                j += p;
            }
        }
    }

    run(&gpu, &pipeline, &sieve, &count, 0, 0, LIMIT, wgpu_common::groups(LIMIT + 1));
    for p in 2..=sqrt {
        if is_prime[p as usize] {
            let marks = (LIMIT - p * p) / p + 1;
            run(&gpu, &pipeline, &sieve, &count, 1, p, LIMIT, wgpu_common::groups(marks));
        }
    }
    run(&gpu, &pipeline, &sieve, &count, 2, 0, LIMIT, wgpu_common::groups(LIMIT + 1));
    let n = gpu.read_u32s(&count, 1)[0];
    println!("Primes up to {}: {}", LIMIT, n);
}

fn run(
    gpu: &wgpu_common::Gpu,
    pipeline: &wgpu::ComputePipeline,
    sieve: &wgpu::Buffer,
    count: &wgpu::Buffer,
    mode: u32,
    p: u32,
    limit: u32,
    groups: u32,
) {
    let ubo = gpu.uniform(bytemuck::bytes_of(&[mode, p, limit, 0u32]));
    let bind = gpu.bind(pipeline, &[sieve, count, &ubo]);
    gpu.dispatch(pipeline, &bind, groups);
}

sorting

Source: examples/wgpu/sorting.rs

const SHADER: &str = r#"
struct Params {
    n: u32,
    j: u32,
    k: u32,
    _pad: u32,
}

@group(0) @binding(0) var<storage, read_write> data: array<u32>;
@group(0) @binding(1) var<uniform> params: Params;

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid: vec3<u32>) {
    let i = gid.x;
    let n = params.n;
    if i >= n {
        return;
    }
    let ixj = i ^ params.j;
    if ixj > i && ixj < n {
        let a = data[i];
        let b = data[ixj];
        let ascending = (i & params.k) == 0u;
        if (a > b) == ascending {
            data[i] = b;
            data[ixj] = a;
        }
    }
}
"#;

fn main() {
    const N: u32 = 1_000_000;
    const PAD: u32 = 1 << 20;
    let gpu = wgpu_common::Gpu::new();

    let mut host = vec![u32::MAX; PAD as usize];
    let mut seed: u32 = 42;
    for i in 0..N as usize {
        seed = seed.wrapping_mul(1664525).wrapping_add(1013904223);
        host[i] = seed & 0x7fff_ffff;
    }

    let buf = gpu.storage_init(bytemuck::cast_slice(&host));
    let pipeline = gpu.pipeline(SHADER);
    let groups = wgpu_common::groups(PAD);

    let mut k = 2u32;
    while k <= PAD {
        let mut j = k / 2;
        while j > 0 {
            let ubo = gpu.uniform(bytemuck::bytes_of(&[PAD, j, k, 0u32]));
            let bind = gpu.bind(&pipeline, &[&buf, &ubo]);
            gpu.dispatch(&pipeline, &bind, groups);
            j /= 2;
        }
        k *= 2;
    }

    let out = gpu.read_u32s(&buf, N as usize);
    println!(
        "Sorted {} integers: first={}, last={}",
        N,
        out[0],
        out[N as usize - 1]
    );
}
