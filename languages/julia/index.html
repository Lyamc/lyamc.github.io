function fib(n)
    a = 0
    b = 1
    for _ in 1:n
        a, b = b, a + b
    end
    a
end
println(fib(42))

function sieve1(limit)
    isprime = trues(limit)
    isprime[1] = false
    lim = isqrt(limit)
    @inbounds for i in 2:lim
        if isprime[i]
            for j in (i * i):i:limit
                isprime[j] = false
            end
        end
    end
    count(isprime)
end

println("Primes up to 1000000: ", sieve1(1_000_000))

function radix_sort!(a::Vector{UInt32})
    n = length(a)
    tmp = Vector{UInt32}(undef, n)
    src, dst = a, tmp
    count = zeros(UInt32, 256)
    @inbounds for pass in 0:3
        fill!(count, 0)
        shift = pass * 8
        for i in 1:n
            count[((src[i] >> shift) & 0xff) + 1] += 1
        end
        sumv = UInt32(0)
        for i in 1:256
            c = count[i]
            count[i] = sumv
            sumv += c
        end
        for i in 1:n
            v = src[i]
            b = Int((v >> shift) & 0xff) + 1
            dst[count[b] + 1] = v
            count[b] += 1
        end
        src, dst = dst, src
    end
    a
end

function lcg(n)
    data = Vector{UInt32}(undef, n)
    seed = UInt32(42)
    @inbounds for i in 1:n
        seed = seed * UInt32(1664525) + UInt32(1013904223)
        data[i] = seed & 0x7fffffff
    end
    data
end

n = 1_000_000
data = radix_sort!(lcg(n))
println("Sorted ", n, " integers: first=", data[1], ", last=", data[end])
