prime_sieve

Source: examples/php/prime_sieve.php

<?php
$limit = 1000000;
$sieve = str_repeat("\x01", $limit + 1);
$sieve[0] = $sieve[1] = "\0";
for ($i = 2; $i * $i <= $limit; $i++) {
    if ($sieve[$i] !== "\0") {
        for ($j = $i * $i; $j <= $limit; $j += $i) {
            $sieve[$j] = "\0";
        }
    }
}
$count = 0;
for ($i = 2; $i <= $limit; $i++) if ($sieve[$i] !== "\0") $count++;
echo "Primes up to $limit: $count\n";

<?php
$n = 1000000;
$data = array_fill(0, $n, 0);
$seed = 42;
for ($i = 0; $i < $n; $i++) {
    $seed = ($seed * 1664525 + 1013904223) & 0xffffffff;
    $data[$i] = $seed & 0x7fffffff;
}
sort($data, SORT_NUMERIC);
echo "Sorted $n integers: first={$data[0]}, last={$data[$n - 1]}\n";
