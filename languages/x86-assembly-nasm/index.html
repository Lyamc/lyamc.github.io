hello

Source: examples/nasm/hello.asm

; Windows x64 hello world
default rel
extern GetStdHandle
extern WriteFile
extern ExitProcess

section .data
msg db "Hello, World!", 13, 10
msg_len equ $ - msg

section .bss
written resd 1

section .text
global main
main:
    sub rsp, 40
    mov ecx, -11                 ; STD_OUTPUT_HANDLE
    call GetStdHandle
    mov rcx, rax
    lea rdx, [msg]
    mov r8d, msg_len
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    xor ecx, ecx
    call ExitProcess

Source: examples/nasm/fibonacci.asm

default rel
extern GetStdHandle
extern WriteFile
extern ExitProcess

section .bss
written resd 1
buf     resb 32

section .text
global main

; print unsigned rax followed by newline
print_u64:
    push rbx
    sub rsp, 32
    lea rdi, [buf+31]
    mov byte [rdi], 10
    dec rdi
    mov byte [rdi], 13
    mov rbx, 10
.conv:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz .conv
    lea rdx, [buf+32]
    sub rdx, rdi
    mov r8, rdx
    mov ecx, -11
    call GetStdHandle
    mov rcx, rax
    mov rdx, rdi
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    add rsp, 32
    pop rbx
    ret

main:
    sub rsp, 40
    mov ecx, 42
    xor rax, rax          ; a = 0
    mov rdx, 1            ; b = 1
.loop:
    test ecx, ecx
    jz .done
    mov r8, rax
    add r8, rdx           ; tmp = a+b
    mov rax, rdx          ; a = b
    mov rdx, r8           ; b = tmp
    dec ecx
    jmp .loop
.done:
    call print_u64
    xor ecx, ecx
    call ExitProcess

prime_sieve

Source: examples/nasm/prime_sieve.asm

default rel
extern GetStdHandle
extern WriteFile
extern ExitProcess
extern GetProcessHeap
extern HeapAlloc
extern HeapFree

LIMIT equ 1000000

section .data
prefix db "Primes up to 1000000: "
prefix_len equ $ - prefix

section .bss
written resd 1
buf     resb 32

section .text
global main

print_u64:
    push rbx
    sub rsp, 32
    lea rdi, [buf+31]
    mov byte [rdi], 10
    dec rdi
    mov byte [rdi], 13
    mov rbx, 10
.conv:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz .conv
    lea rdx, [buf+32]
    sub rdx, rdi
    mov r8, rdx
    mov ecx, -11
    call GetStdHandle
    mov rcx, rax
    mov rdx, rdi
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    add rsp, 32
    pop rbx
    ret

main:
    push rbx
    push rsi
    push rdi
    push r12
    push r13
    sub rsp, 48
    call GetProcessHeap
    mov r12, rax
    mov rcx, r12
    mov edx, 8                ; HEAP_ZERO_MEMORY
    mov r8, LIMIT+1
    call HeapAlloc
    test rax, rax
    jz .fail
    mov r13, rax              ; sieve
    mov rdi, r13
    mov rcx, LIMIT+1
    mov al, 1
    rep stosb
    mov byte [r13], 0
    mov byte [r13+1], 0
    mov esi, 2
.outer:
    mov eax, esi
    imul eax, esi
    cmp eax, LIMIT
    jg .count
    cmp byte [r13+rsi], 0
    je .nexti
    mov ebx, eax
.inner:
    cmp ebx, LIMIT
    jg .nexti
    mov byte [r13+rbx], 0
    add ebx, esi
    jmp .inner
.nexti:
    inc esi
    jmp .outer
.count:
    xor ebx, ebx
    mov esi, 2
.cnt:
    cmp esi, LIMIT
    jg .print
    cmp byte [r13+rsi], 0
    je .cntn
    inc ebx
.cntn:
    inc esi
    jmp .cnt
.print:
    mov ecx, -11
    call GetStdHandle
    mov rcx, rax
    lea rdx, [prefix]
    mov r8d, prefix_len
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    mov eax, ebx
    call print_u64
    mov rcx, r12
    xor edx, edx
    mov r8, r13
    call HeapFree
    xor ecx, ecx
    call ExitProcess
.fail:
    mov ecx, 1
    call ExitProcess

sorting

Source: examples/nasm/sorting.asm

default rel
extern GetStdHandle
extern WriteFile
extern ExitProcess
extern GetProcessHeap
extern HeapAlloc
extern HeapFree

N equ 1000000

section .data
p1 db "Sorted 1000000 integers: first="
p1_len equ $ - p1
p2 db ", last="
p2_len equ $ - p2

section .bss
written resd 1
buf     resb 32
counts  resd 256

section .text
global main

print_u64_raw:
    push rbx
    sub rsp, 32
    lea rdi, [buf+31]
    mov byte [rdi], 0
    mov rbx, 10
.conv:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz .conv
    lea rdx, [buf+32]
    sub rdx, rdi
    dec rdx
    mov r8, rdx
    mov ecx, -11
    call GetStdHandle
    mov rcx, rax
    mov rdx, rdi
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    add rsp, 32
    pop rbx
    ret

write_str:
    ; rcx unused, rdx=ptr, r8=len
    push r8
    push rdx
    sub rsp, 40
    mov ecx, -11
    call GetStdHandle
    add rsp, 40
    pop rdx
    pop r8
    sub rsp, 40
    mov rcx, rax
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    add rsp, 40
    ret

; LSD radix sort, 4 passes of 8 bits, 256 buckets
; r12 = src (starts as data), r13 = dst (scratch)
; after 4 pointer swaps, r12 holds the sorted array
; %1 = byte offset 0..3 (little-endian)
%macro RADIX_PASS 1
    lea rdi, [rel counts]
    mov ecx, 256
    xor eax, eax
    cld
    rep stosd
    lea r8, [rel counts]
    mov rsi, r12
    lea r9, [r12+N*4]
%%hist:
    movzx eax, byte [rsi+%1]
    add rsi, 4
    inc dword [r8+rax*4]
    cmp rsi, r9
    jne %%hist
    xor eax, eax
    xor ebx, ebx
%%prefix:
    mov edx, [r8+rbx*4]
    mov [r8+rbx*4], eax
    add eax, edx
    inc ebx
    cmp ebx, 256
    jb %%prefix
    mov rsi, r12
    lea r9, [r12+N*4]
%%scatter:
    mov eax, [rsi]
    movzx ebx, byte [rsi+%1]
    add rsi, 4
    mov edx, [r8+rbx*4]
    mov [r13+rdx*4], eax
    inc edx
    mov [r8+rbx*4], edx
    cmp rsi, r9
    jne %%scatter
    xchg r12, r13
%endmacro

radix_sort:
    push rbx
    push rsi
    push rdi
    RADIX_PASS 0
    RADIX_PASS 1
    RADIX_PASS 2
    RADIX_PASS 3
    pop rdi
    pop rsi
    pop rbx
    ret

main:
    push rbx
    push rsi
    push rdi
    push r12
    push r13
    push r14
    push r15
    sub rsp, 32
    call GetProcessHeap
    mov r14, rax
    mov rcx, r14
    xor edx, edx
    mov r8, N*4
    call HeapAlloc
    test rax, rax
    jz .fail
    mov r12, rax
    mov rcx, r14
    xor edx, edx
    mov r8, N*4
    call HeapAlloc
    test rax, rax
    jz .fail
    mov r13, rax
    mov r15, r12                    ; keep original data pointer for free
    xor esi, esi
    mov edi, 42                     ; seed
.fill:
    imul edi, edi, 1664525
    add edi, 1013904223
    mov eax, edi
    and eax, 0x7fffffff
    mov [r12+rsi*4], eax
    inc esi
    cmp esi, N
    jb .fill
    call radix_sort
    lea rdx, [p1]
    mov r8d, p1_len
    call write_str
    mov eax, [r12]
    call print_u64_raw
    lea rdx, [p2]
    mov r8d, p2_len
    call write_str
    mov eax, [r12+(N-1)*4]
    call print_u64_raw
    mov ecx, -11
    call GetStdHandle
    mov rcx, rax
    mov byte [buf], 13
    mov byte [buf+1], 10
    lea rdx, [buf]
    mov r8d, 2
    lea r9, [written]
    mov qword [rsp+32], 0
    call WriteFile
    mov rcx, r14
    xor edx, edx
    mov r8, r15
    call HeapFree
    mov rcx, r14
    xor edx, edx
    ; r13 is the other buffer after even swaps (r12=data, r13=scratch)
    mov r8, r13
    call HeapFree
    xor ecx, ecx
    call ExitProcess
.fail:
    mov ecx, 1
    call ExitProcess
