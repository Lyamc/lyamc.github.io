prime_sieve

Source: examples/powershell/prime_sieve.ps1

# PS 5.1 loops over 1e6 are the old 2.5s outlier. Same Add-Type pattern as sorting.
Add-Type -TypeDefinition @"
public static class Sieve {
    public static int Count(int limit) {
        bool[] comp = new bool[limit + 1];
        for (int i = 2; i * i <= limit; i++) {
            if (!comp[i]) {
                for (int j = i * i; j <= limit; j += i) comp[j] = true;
            }
        }
        int count = 0;
        for (int i = 2; i <= limit; i++) if (!comp[i]) count++;
        return count;
    }
}
"@
$limit = 1000000
$count = [Sieve]::Count($limit)
Write-Output "Primes up to ${limit}: $count"

sorting

Source: examples/powershell/sorting.ps1

# PS 5.1 promotes uint32 multiply to Double. Do fill+radix in C#.
Add-Type -TypeDefinition @"
public static class LcgRadix {
    static void RadixSort(int[] a) {
        int n = a.Length;
        int[] tmp = new int[n];
        int[] src = a, dst = tmp;
        int[] count = new int[256];
        for (int pass = 0; pass < 4; pass++) {
            System.Array.Clear(count, 0, 256);
            int shift = pass * 8;
            for (int i = 0; i < n; i++) count[(src[i] >> shift) & 255]++;
            int sum = 0;
            for (int i = 0; i < 256; i++) {
                int c = count[i];
                count[i] = sum;
                sum += c;
            }
            for (int i = 0; i < n; i++) {
                int v = src[i];
                int b = (v >> shift) & 255;
                dst[count[b]++] = v;
            }
            int[] t = src; src = dst; dst = t;
        }
        if (src != a) System.Array.Copy(src, a, n);
    }
    public static string Run(int n) {
        int[] data = new int[n];
        uint seed = 42;
        for (int i = 0; i < n; i++) {
            seed = seed * 1664525u + 1013904223u;
            data[i] = (int)(seed & 0x7fffffffu);
        }
        RadixSort(data);
        return "Sorted " + n + " integers: first=" + data[0] + ", last=" + data[n - 1];
    }
}
"@
Write-Output ([LcgRadix]::Run(1000000))
