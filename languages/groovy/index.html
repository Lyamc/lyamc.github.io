prime_sieve

Source: examples/groovy/prime_sieve.groovy

def limit = 1_000_000
def sieve = new boolean[limit + 1]
Arrays.fill(sieve, true)
sieve[0] = sieve[1] = false
for (int i = 2; i * i <= limit; i++) {
    if (sieve[i]) {
        for (int j = i * i; j <= limit; j += i) sieve[j] = false
    }
}
def count = 0
for (int i = 2; i <= limit; i++) if (sieve[i]) count++
println "Primes up to ${limit}: ${count}"

Source: examples/groovy/sorting.groovy

def n = 1_000_000
def data = new int[n]
def seed = 42
for (int i = 0; i < n; i++) {
    seed = seed * 1664525 + 1013904223
    data[i] = seed & 0x7fffffff
}
Arrays.sort(data)
println "Sorted ${n} integers: first=${data[0]}, last=${data[n - 1]}"
