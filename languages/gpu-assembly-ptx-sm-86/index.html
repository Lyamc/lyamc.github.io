hello

Source: examples/ptx/hello.ptx

// Handwritten PTX for NVIDIA sm_86 (Ampere / GeForce RTX 3060)
.version 8.5
.target sm_86
.address_size 64

// Thread 0 writes "Hello, World!\n\0" (15 bytes) to global memory.
.visible .entry hello(.param .u64 out) {
    .reg .pred %p<2>;
    .reg .b32 %r<8>;
    .reg .b64 %rd<4>;

    mov.u32     %r0, %tid.x;
    mov.u32     %r1, %ctaid.x;
    or.b32      %r2, %r0, %r1;
    setp.ne.u32 %p0, %r2, 0;
    @%p0        ret;

    ld.param.u64 %rd0, [out];

    // "Hell" "o, W" "orld" "!\n\0"
    mov.u32     %r3, 0x6c6c6548;
    st.global.u32 [%rd0], %r3;
    mov.u32     %r3, 0x57202c6f;
    st.global.u32 [%rd0+4], %r3;
    mov.u32     %r3, 0x646c726f;
    st.global.u32 [%rd0+8], %r3;
    mov.u32     %r3, 0x00000a21;
    st.global.u32 [%rd0+12], %r3;
    ret;
}

fibonacci

Source: examples/ptx/fibonacci.ptx

// Handwritten PTX for NVIDIA sm_86 (Ampere / GeForce RTX 3060)
// Iterative F(42); result in u32 (267914296).
.version 8.5
.target sm_86
.address_size 64

.visible .entry fibonacci(
    .param .u64 result,
    .param .u32 n
) {
    .reg .pred %p<2>;
    .reg .b32 %r<8>;
    .reg .b64 %rd<2>;

    ld.param.u64 %rd0, [result];
    ld.param.u32 %r3, [n];

    mov.u32     %r0, 0;          // a
    mov.u32     %r1, 1;          // b
    mov.u32     %r2, 0;          // i

loop:
    setp.ge.u32 %p0, %r2, %r3;
    @%p0        bra done;
    add.u32     %r4, %r0, %r1;   // tmp = a + b
    mov.u32     %r0, %r1;        // a = b
    mov.u32     %r1, %r4;        // b = tmp
    add.u32     %r2, %r2, 1;
    bra.uni     loop;

done:
    st.global.u32 [%rd0], %r0;
    ret;
}

prime_sieve

Source: examples/ptx/prime_sieve.ptx

// Handwritten PTX for NVIDIA sm_86 (Ampere / GeForce RTX 3060)
// Sieve of Eratosthenes: init / mark multiples / atomic count.
.version 8.5
.target sm_86
.address_size 64

.visible .entry sieve_init(
    .param .u64 sieve,
    .param .u32 limit
) {
    .reg .pred %p<3>;
    .reg .b32 %r<8>;
    .reg .b64 %rd<4>;

    ld.param.u64 %rd0, [sieve];
    ld.param.u32 %r4, [limit];

    mov.u32     %r0, %tid.x;
    mov.u32     %r1, %ntid.x;
    mov.u32     %r2, %ctaid.x;
    mad.lo.u32  %r3, %r2, %r1, %r0;   // i

    setp.gt.u32 %p0, %r3, %r4;
    @%p0        ret;

    setp.lt.u32 %p1, %r3, 2;
    selp.u32    %r5, 0, 1, %p1;

    mul.wide.u32 %rd1, %r3, 4;
    add.u64     %rd1, %rd0, %rd1;
    st.global.u32 [%rd1], %r5;
    ret;
}

// Mark sieve[p*p + tid*p] = 0
.visible .entry sieve_mark(
    .param .u64 sieve,
    .param .u32 p,
    .param .u32 limit
) {
    .reg .pred %p<2>;
    .reg .b32 %r<12>;
    .reg .b64 %rd<4>;

    ld.param.u64 %rd0, [sieve];
    ld.param.u32 %r5, [p];
    ld.param.u32 %r6, [limit];

    mov.u32     %r0, %tid.x;
    mov.u32     %r1, %ntid.x;
    mov.u32     %r2, %ctaid.x;
    mad.lo.u32  %r3, %r2, %r1, %r0;   // tid

    mul.lo.u32  %r7, %r5, %r5;        // p*p
    mad.lo.u32  %r8, %r3, %r5, %r7;   // j = tid*p + p*p

    setp.gt.u32 %p0, %r8, %r6;
    @%p0        ret;

    mul.wide.u32 %rd1, %r8, 4;
    add.u64     %rd1, %rd0, %rd1;
    mov.u32     %r9, 0;
    st.global.u32 [%rd1], %r9;
    ret;
}

.visible .entry sieve_count(
    .param .u64 sieve,
    .param .u64 count,
    .param .u32 limit
) {
    .reg .pred %p<2>;
    .reg .b32 %r<8>;
    .reg .b64 %rd<4>;

    ld.param.u64 %rd0, [sieve];
    ld.param.u64 %rd2, [count];
    ld.param.u32 %r4, [limit];

    mov.u32     %r0, %tid.x;
    mov.u32     %r1, %ntid.x;
    mov.u32     %r2, %ctaid.x;
    mad.lo.u32  %r3, %r2, %r1, %r0;   // i

    setp.gt.u32 %p0, %r3, %r4;
    @%p0        ret;

    mul.wide.u32 %rd1, %r3, 4;
    add.u64     %rd1, %rd0, %rd1;
    ld.global.u32 %r5, [%rd1];
    setp.eq.u32 %p1, %r5, 1;
    @%p1        atom.global.add.u32 %r6, [%rd2], 1;
    ret;
}

sorting

Source: examples/ptx/sorting.ptx

// Handwritten PTX for NVIDIA sm_86 (Ampere / GeForce RTX 3060)
// One bitonic-compare stage. Host loops k=2..n, j=k/2..1.
.version 8.5
.target sm_86
.address_size 64

.visible .entry bitonic(
    .param .u64 data,
    .param .u32 n,
    .param .u32 j,
    .param .u32 k
) {
    .reg .pred %p<6>;
    .reg .b32 %r<16>;
    .reg .b64 %rd<6>;

    ld.param.u64 %rd0, [data];
    ld.param.u32 %r10, [n];
    ld.param.u32 %r11, [j];
    ld.param.u32 %r12, [k];

    mov.u32     %r0, %tid.x;
    mov.u32     %r1, %ntid.x;
    mov.u32     %r2, %ctaid.x;
    mad.lo.u32  %r3, %r2, %r1, %r0;   // i

    setp.ge.u32 %p0, %r3, %r10;
    @%p0        ret;

    xor.b32     %r4, %r3, %r11;       // ixj = i ^ j
    setp.le.u32 %p1, %r4, %r3;        // only the lower index of the pair
    @%p1        ret;
    setp.ge.u32 %p2, %r4, %r10;
    @%p2        ret;

    mul.wide.u32 %rd1, %r3, 4;
    add.u64     %rd1, %rd0, %rd1;
    mul.wide.u32 %rd2, %r4, 4;
    add.u64     %rd2, %rd0, %rd2;

    ld.global.u32 %r5, [%rd1];        // a
    ld.global.u32 %r6, [%rd2];        // b

    and.b32     %r7, %r3, %r12;
    setp.eq.u32 %p3, %r7, 0;          // ascending = (i & k) == 0
    setp.gt.u32 %p4, %r5, %r6;        // a > b
    xor.pred    %p5, %p3, %p4;        // skip swap if (a>b) != ascending
    @%p5        ret;

    st.global.u32 [%rd1], %r6;
    st.global.u32 [%rd2], %r5;
    ret;
}
