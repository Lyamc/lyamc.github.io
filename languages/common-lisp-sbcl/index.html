(defun fib (n)
  (let ((a 0) (b 1))
    (dotimes (i n)
      (let ((tmp (+ a b)))
        (setf a b b tmp)))
    a))
(format t "~A~%" (fib 42))

Source: examples/lisp/prime_sieve.lisp

(defun sieve (limit)
  (let ((is-prime (make-array (1+ limit) :element-type 'bit :initial-element 1)))
    (setf (sbit is-prime 0) 0
          (sbit is-prime 1) 0)
    (loop for i from 2 to (isqrt limit)
          when (= (sbit is-prime i) 1)
            do (loop for j from (* i i) to limit by i
                     do (setf (sbit is-prime j) 0)))
    (loop for i from 2 to limit count (= (sbit is-prime i) 1))))
(format t "Primes up to 1000000: ~A~%" (sieve 1000000))

Source: examples/lisp/sorting.lisp

(declaim (optimize (speed 3) (safety 0) (debug 0)))

(deftype u32 () '(unsigned-byte 32))
(deftype u32vec () '(simple-array u32 (*)))
(deftype idx () '(integer 0 10000000))

(defun lcg-fill (n)
  (declare (type idx n))
  (let ((data (make-array n :element-type 'u32))
        (seed 42))
    (declare (type u32vec data) (type u32 seed))
    (dotimes (i n)
      (declare (type idx i))
      (setf seed (ldb (byte 32 0) (+ (* seed 1664525) 1013904223))
            (aref data i) (logand seed #x7fffffff)))
    data))

(defun radix-sort (a)
  (declare (type u32vec a))
  (let* ((n (length a))
         (tmp (make-array n :element-type 'u32))
         (count (make-array 256 :element-type 'u32 :initial-element 0)))
    (declare (type u32vec tmp) (type (simple-array u32 (256)) count) (type idx n))
    (let ((src a) (dst tmp))
      (declare (type u32vec src dst))
      (dotimes (pass 4)
        (fill count 0)
        (let ((shift (* pass 8)))
          (declare (type (integer 0 24) shift))
          (dotimes (i n)
            (let ((b (ldb (byte 8 shift) (aref src i))))
              (incf (aref count b))))
          (let ((sum 0))
            (declare (type u32 sum))
            (dotimes (i 256)
              (let ((c (aref count i)))
                (setf (aref count i) sum)
                (incf sum c))))
          (dotimes (i n)
            (let* ((v (aref src i))
                   (b (ldb (byte 8 shift) v)))
              (setf (aref dst (aref count b)) v)
              (incf (aref count b)))))
        (rotatef src dst))
      a)))

(let ((data (radix-sort (lcg-fill 1000000))))
  (format t "Sorted 1000000 integers: first=~A, last=~A~%"
          (aref data 0)
          (aref data 999999)))
