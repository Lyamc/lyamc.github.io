prime_sieve

Source: examples/sql/prime_sieve.sql

-- Sieve of Eratosthenes up to 1_000_000:
-- generate 2..N, then mark composites as p^2, p^2+p, p^2+2p, ...
DROP TABLE IF EXISTS nums;
CREATE TABLE nums(n INTEGER PRIMARY KEY, is_prime INTEGER NOT NULL);

WITH RECURSIVE seq(n) AS (
  SELECT 2
  UNION ALL
  SELECT n + 1 FROM seq WHERE n < 1000000
)
INSERT INTO nums SELECT n, 1 FROM seq;

WITH RECURSIVE
small(n) AS (
  SELECT 2
  UNION ALL
  SELECT n + 1 FROM small WHERE n < 1000
),
small_mark(p, m) AS (
  SELECT n, n * n FROM small WHERE n * n <= 1000
  UNION ALL
  SELECT p, m + p FROM small_mark WHERE m + p <= 1000
),
prime_p(p) AS (
  SELECT s.n
  FROM small s
  LEFT JOIN small_mark sm ON s.n = sm.m
  WHERE sm.m IS NULL
),
composites(p, m) AS (
  SELECT p, p * p FROM prime_p
  UNION ALL
  SELECT p, m + p FROM composites WHERE m + p <= 1000000
)
UPDATE nums SET is_prime = 0
 WHERE n IN (SELECT m FROM composites);

SELECT 'Primes up to 1000000: ' || COUNT(*) FROM nums WHERE is_prime = 1;

sorting

Source: examples/sql/sorting.sql

DROP TABLE IF EXISTS vals;
DROP TABLE IF EXISTS sorted;
CREATE TABLE vals(v INTEGER);
WITH RECURSIVE seq(i, seed) AS (
  SELECT 0, (42 * 1664525 + 1013904223) & 4294967295
  UNION ALL
  SELECT i + 1, (seed * 1664525 + 1013904223) & 4294967295 FROM seq WHERE i < 999999
)
INSERT INTO vals SELECT (seed & 2147483647) FROM seq;
CREATE TABLE sorted AS SELECT v FROM vals ORDER BY v;
SELECT 'Sorted 1000000 integers: first=' || (SELECT v FROM sorted WHERE rowid = 1)
  || ', last=' || (SELECT v FROM sorted WHERE rowid = 1000000);
