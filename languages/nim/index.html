prime_sieve

Source: examples/nim/prime_sieve.nim

const limit = 1_000_000
var sieve = newSeq[byte](limit + 1)
for i in 0..limit:
  sieve[i] = 1
sieve[0] = 0
sieve[1] = 0
var i = 2
while i * i <= limit:
  if sieve[i] != 0:
    var j = i * i
    while j <= limit:
      sieve[j] = 0
      j += i
  inc i
var count = 0
for k in 2..limit:
  if sieve[k] != 0: inc count
echo "Primes up to ", limit, ": ", count

sorting

Source: examples/nim/sorting.nim

const n = 1_000_000

proc radixSort(a: var seq[uint32]) =
  var tmp = newSeq[uint32](a.len)
  var src = cast[ptr UncheckedArray[uint32]](addr a[0])
  var dst = cast[ptr UncheckedArray[uint32]](addr tmp[0])
  for pass in 0..3:
    var count: array[256, uint32]
    let shift = uint32(pass * 8)
    for i in 0..<a.len:
      inc count[int((src[i] shr shift) and 255'u32)]
    var sum: uint32 = 0
    for i in 0..255:
      let c = count[i]
      count[i] = sum
      sum += c
    for i in 0..<a.len:
      let v = src[i]
      let b = int((v shr shift) and 255'u32)
      dst[count[b]] = v
      inc count[b]
    swap src, dst

var data = newSeq[uint32](n)
var seed: uint32 = 42
for i in 0..<n:
  seed = seed * 1664525'u32 + 1013904223'u32
  data[i] = seed and 0x7fffffff'u32
radixSort(data)
echo "Sorted ", n, " integers: first=", data[0], ", last=", data[^1]
