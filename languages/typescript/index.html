Source: examples/typescript/prime_sieve.ts

const LIMIT = 1_000_000;
const sieve = new Uint8Array(LIMIT + 1);
sieve.fill(1, 2);
for (let i = 2; i * i <= LIMIT; i++) {
  if (sieve[i]) {
    for (let j = i * i; j <= LIMIT; j += i) sieve[j] = 0;
  }
}
let count = 0;
for (let i = 2; i <= LIMIT; i++) if (sieve[i]) count++;
console.log(`Primes up to ${LIMIT}: ${count}`);

Source: examples/typescript/sorting.ts

function radixSort(a: Int32Array): void {
  const n = a.length;
  const tmp = new Int32Array(n);
  let src = a, dst = tmp;
  const count = new Uint32Array(256);
  for (let pass = 0; pass < 4; pass++) {
    count.fill(0);
    const shift = pass * 8;
    for (let i = 0; i < n; i++) count[(src[i] >>> shift) & 255]++;
    let sum = 0;
    for (let i = 0; i < 256; i++) {
      const c = count[i];
      count[i] = sum;
      sum += c;
    }
    for (let i = 0; i < n; i++) {
      const v = src[i];
      const b = (v >>> shift) & 255;
      dst[count[b]++] = v;
    }
    const t = src;
    src = dst;
    dst = t;
  }
}

const N = 1_000_000;
let seed = 42;
const data = new Int32Array(N);
for (let i = 0; i < N; i++) {
  seed = (Math.imul(seed, 1664525) + 1013904223) | 0;
  data[i] = seed & 0x7fffffff;
}
radixSort(data);
console.log(`Sorted ${N} integers: first=${data[0]}, last=${data[N - 1]}`);
