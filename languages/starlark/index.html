Source: examples/starlark/prime_sieve.star

def sieve():
    limit = 1000000
    flags = [True] * (limit + 1)
    flags[0] = False
    flags[1] = False
    for i in range(2, 1001):
        if flags[i]:
            j = i * i
            for k in range(j, limit + 1, i):
                flags[k] = False
    count = 0
    for i in range(2, limit + 1):
        if flags[i]:
            count += 1
    print("Primes up to %d: %d" % (limit, count))

sieve()

def sort_run():
    n = 1000000
    seed = 42
    data = [0] * n
    for i in range(n):
        seed = (seed * 1664525 + 1013904223) & 0xffffffff
        data[i] = seed & 0x7fffffff
    data = sorted(data)
    print("Sorted %d integers: first=%d, last=%d" % (n, data[0], data[n - 1]))
