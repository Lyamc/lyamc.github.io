prime_sieve

Source: examples/swift/prime_sieve.swift

let limit = 1_000_000
// Odd numbers only: index i represents 2 * i + 1.
let size = (limit + 1) >> 1
let sieve = UnsafeMutablePointer<UInt8>.allocate(capacity: size)
defer { sieve.deallocate() }
sieve.initialize(repeating: 1, count: size)
sieve[0] = 0
var i = 1
while true {
    let p = i &<< 1 &+ 1
    if p * p > limit { break }
    if sieve[i] != 0 {
        var j = (p * p) >> 1
        while j < size {
            sieve[j] = 0
            j &+= p
        }
    }
    i &+= 1
}
var count = 1
var k = 1
while k < size {
    if sieve[k] != 0 { count += 1 }
    k &+= 1
}
print("Primes up to \(limit): \(count)")

sorting

Source: examples/swift/sorting.swift

let n = 1_000_000
let data = UnsafeMutablePointer<UInt32>.allocate(capacity: n)
let tmp = UnsafeMutablePointer<UInt32>.allocate(capacity: n)
defer {
    data.deallocate()
    tmp.deallocate()
}
var seed: UInt32 = 42
for i in 0..<n {
    seed = seed &* 1664525 &+ 1013904223
    data[i] = seed & 0x7fffffff
}
var src = data
var dst = tmp
for pass in 0..<4 {
    var count = [UInt32](repeating: 0, count: 256)
    let shift = pass * 8
    for i in 0..<n {
        count[Int((src[i] >> shift) & 255)] += 1
    }
    var sum: UInt32 = 0
    for i in 0..<256 {
        let c = count[i]
        count[i] = sum
        sum += c
    }
    for i in 0..<n {
        let v = src[i]
        let b = Int((v >> shift) & 255)
        dst[Int(count[b])] = v
        count[b] += 1
    }
    swap(&src, &dst)
}
print("Sorted \(n) integers: first=\(data[0]), last=\(data[n - 1])")
