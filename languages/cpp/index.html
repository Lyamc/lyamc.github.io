prime_sieve

Source: examples/cpp/prime_sieve.cpp

#include <iostream>
#include <vector>

int main() {
    const int LIMIT = 1000000;
    std::vector<char> sieve(LIMIT + 1, 1);
    sieve[0] = sieve[1] = 0;
    for (int i = 2; i * i <= LIMIT; ++i) {
        if (sieve[i]) {
            for (int j = i * i; j <= LIMIT; j += i)
                sieve[j] = 0;
        }
    }
    int count = 0;
    for (int i = 2; i <= LIMIT; ++i)
        if (sieve[i]) ++count;
    std::cout << "Primes up to " << LIMIT << ": " << count << '\n';
    return 0;
}

Source: examples/cpp/sorting.cpp

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void radix_sort(uint32_t *a, uint32_t *tmp, int n) {
    uint32_t count[256];
    uint32_t *src = a, *dst = tmp;
    for (int pass = 0; pass < 4; pass++) {
        memset(count, 0, sizeof count);
        int shift = pass * 8;
        for (int i = 0; i < n; i++)
            count[(src[i] >> shift) & 255]++;
        uint32_t sum = 0;
        for (int i = 0; i < 256; i++) {
            uint32_t c = count[i];
            count[i] = sum;
            sum += c;
        }
        for (int i = 0; i < n; i++) {
            uint32_t v = src[i];
            dst[count[(v >> shift) & 255]++] = v;
        }
        uint32_t *t = src;
        src = dst;
        dst = t;
    }
}

int main() {
    const int N = 1000000;
    uint32_t *data = (uint32_t *)malloc(N * sizeof(uint32_t));
    uint32_t *tmp = (uint32_t *)malloc(N * sizeof(uint32_t));
    if (!data || !tmp) return 1;
    uint32_t seed = 42;
    for (int i = 0; i < N; ++i) {
        seed = seed * 1664525u + 1013904223u;
        data[i] = seed & 0x7fffffffu;
    }
    radix_sort(data, tmp, N);
    printf("Sorted %d integers: first=%u, last=%u\n", N, data[0], data[N - 1]);
    free(tmp);
    free(data);
    return 0;
}
