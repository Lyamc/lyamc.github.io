prime_sieve

Source: examples/objc/prime_sieve.m

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int main(void) {
    const int LIMIT = 1000000;
    char *sieve = malloc((size_t)LIMIT + 1);
    memset(sieve, 1, (size_t)LIMIT + 1);
    sieve[0] = sieve[1] = 0;
    for (int i = 2; i * i <= LIMIT; i++)
        if (sieve[i])
            for (int j = i * i; j <= LIMIT; j += i)
                sieve[j] = 0;
    int count = 0;
    for (int i = 2; i <= LIMIT; i++) if (sieve[i]) count++;
    printf("Primes up to %d: %d\n", LIMIT, count);
    free(sieve);
    return 0;
}

sorting

Source: examples/objc/sorting.m

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void radix_sort(unsigned *a, unsigned *tmp, int n) {
    unsigned count[256];
    unsigned *src = a, *dst = tmp;
    for (int pass = 0; pass < 4; pass++) {
        memset(count, 0, sizeof count);
        int shift = pass * 8;
        for (int i = 0; i < n; i++)
            count[(src[i] >> shift) & 255]++;
        unsigned sum = 0;
        for (int i = 0; i < 256; i++) {
            unsigned c = count[i];
            count[i] = sum;
            sum += c;
        }
        for (int i = 0; i < n; i++) {
            unsigned v = src[i];
            dst[count[(v >> shift) & 255]++] = v;
        }
        unsigned *t = src;
        src = dst;
        dst = t;
    }
}

int main(void) {
    const int N = 1000000;
    unsigned *data = malloc((size_t)N * sizeof *data);
    unsigned *tmp = malloc((size_t)N * sizeof *tmp);
    if (!data || !tmp) return 1;
    unsigned seed = 42;
    for (int i = 0; i < N; i++) {
        seed = seed * 1664525u + 1013904223u;
        data[i] = seed & 0x7fffffffu;
    }
    radix_sort(data, tmp, N);
    printf("Sorted %d integers: first=%u, last=%u\n", N, data[0], data[N - 1]);
    free(tmp);
    free(data);
    return 0;
}
