prime_sieve

Source: examples/ruby/prime_sieve.rb

LIMIT = 1_000_000
sieve = "\x01" * (LIMIT + 1)
sieve.setbyte(0, 0)
sieve.setbyte(1, 0)
i = 2
while i * i <= LIMIT
  if sieve.getbyte(i) != 0
    j = i * i
    while j <= LIMIT
      sieve.setbyte(j, 0)
      j += i
    end
  end
  i += 1
end
count = sieve.count("\x01")
puts "Primes up to #{LIMIT}: #{count}"
