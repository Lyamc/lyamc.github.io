    var b: i64 = 1;

        const tmp = a + b;
        a = b;
        b = tmp;
    }

prime_sieve

Source: examples/zig/prime_sieve.zig

const std = @import("std");

pub fn main() !void {
    const limit: usize = 1_000_000;
    const alloc = std.heap.page_allocator;
    const sieve = try alloc.alloc(u8, limit + 1);
    defer alloc.free(sieve);
    @memset(sieve, 1);
    sieve[0] = 0;
    sieve[1] = 0;
    var i: usize = 2;
    while (i * i <= limit) : (i += 1) {
        if (sieve[i] != 0) {
            var j = i * i;
            while (j <= limit) : (j += i) {
                sieve[j] = 0;
            }
        }
    }
    var count: usize = 0;
    i = 2;
    while (i <= limit) : (i += 1) {
        if (sieve[i] != 0) count += 1;
    }
    std.debug.print("Primes up to {d}: {d}\n", .{ limit, count });
}

Source: examples/zig/sorting.zig

const std = @import("std");

fn radixSort(a: []u32, tmp: []u32) void {
    var src = a;
    var dst = tmp;
    var pass: u5 = 0;
    while (pass < 4) : (pass += 1) {
        var count = [_]u32{0} ** 256;
        const shift: u5 = pass * 8;
        for (src) |v| {
            count[@as(usize, @intCast(v >> shift)) & 255] += 1;
        }
        var sum: u32 = 0;
        for (&count) |*c| {
            const old = c.*;
            c.* = sum;
            sum += old;
        }
        for (src) |v| {
            const b = @as(usize, @intCast(v >> shift)) & 255;
            dst[count[b]] = v;
            count[b] += 1;
        }
        const t = src;
        src = dst;
        dst = t;
    }
}

pub fn main() !void {
    const n: usize = 1_000_000;
    const alloc = std.heap.page_allocator;
    const data = try alloc.alloc(u32, n);
    defer alloc.free(data);
    const tmp = try alloc.alloc(u32, n);
    defer alloc.free(tmp);
    var seed: u32 = 42;
    for (data) |*x| {
        seed = seed *% 1664525 +% 1013904223;
        x.* = seed & 0x7fffffff;
    }
    radixSort(data, tmp);
    std.debug.print("Sorted {d} integers: first={d}, last={d}\n", .{ n, data[0], data[n - 1] });
}
