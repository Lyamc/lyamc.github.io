prime_sieve

Source: examples/d/prime_sieve.d

import std.stdio;

void main() {
    enum limit = 1_000_000;
    auto sieve = new bool[limit + 1];
    sieve[] = true;
    sieve[0] = sieve[1] = false;
    for (int i = 2; i * i <= limit; ++i) {
        if (sieve[i]) {
            for (int j = i * i; j <= limit; j += i)
                sieve[j] = false;
        }
    }
    int count = 0;
    foreach (i; 2 .. limit + 1)
        if (sieve[i]) ++count;
    writefln("Primes up to %s: %s", limit, count);
}

sorting

Source: examples/d/sorting.d

import core.stdc.stdio : printf;
import core.stdc.stdlib : malloc, free;
import core.stdc.string : memset;

extern (C) void main() {
    enum n = 1_000_000;
    auto data = cast(uint*) malloc(n * uint.sizeof);
    auto tmp = cast(uint*) malloc(n * uint.sizeof);
    if (data is null || tmp is null) return;
    uint seed = 42;
    for (int i = 0; i < n; i++) {
        seed = seed * 1664525 + 1013904223;
        data[i] = seed & 0x7fffffff;
    }
    uint[256] count;
    auto src = data;
    auto dst = tmp;
    for (int pass = 0; pass < 4; pass++) {
        memset(count.ptr, 0, count.sizeof);
        immutable shift = pass * 8;
        for (int i = 0; i < n; i++)
            count[(src[i] >> shift) & 255]++;
        uint sum = 0;
        for (int i = 0; i < 256; i++) {
            auto c = count[i];
            count[i] = sum;
            sum += c;
        }
        for (int i = 0; i < n; i++) {
            auto v = src[i];
            dst[count[(v >> shift) & 255]++] = v;
        }
        auto t = src;
        src = dst;
        dst = t;
    }
    printf("Sorted %d integers: first=%u, last=%u\n", n, data[0], data[n - 1]);
    free(tmp);
    free(data);
}
