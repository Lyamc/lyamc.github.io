prime_sieve

Source: examples/crystal/prime_sieve.cr

limit = 1_000_000
sieve = Array.new(limit + 1, true)
sieve[0] = sieve[1] = false
i = 2
while i * i <= limit
  if sieve[i]
    j = i * i
    while j <= limit
      sieve[j] = false
      j += i
    end
  end
  i += 1
end
count = 0
(2..limit).each { |k| count += 1 if sieve[k] }
puts "Primes up to #{limit}: #{count}"

sorting

Source: examples/crystal/sorting.cr

n = 1_000_000
data = Slice(UInt32).new(n)
tmp = Slice(UInt32).new(n)
seed = 42_u32
n.times do |i|
  seed = seed &* 1664525_u32 &+ 1013904223_u32
  data[i] = seed & 0x7fffffff_u32
end
src, dst = data, tmp
4.times do |pass|
  count = StaticArray(UInt32, 256).new(0_u32)
  shift = pass * 8
  n.times { |i| count[(src[i] >> shift) & 255] += 1 }
  sum = 0_u32
  256.times do |i|
    c = count[i]
    count[i] = sum
    sum += c
  end
  n.times do |i|
    v = src[i]
    b = (v >> shift) & 255
    dst[count[b]] = v
    count[b] += 1
  end
  src, dst = dst, src
end
puts "Sorted #{n} integers: first=#{data[0]}, last=#{data[n - 1]}"
