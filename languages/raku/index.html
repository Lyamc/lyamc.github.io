prime_sieve

Source: examples/raku/prime_sieve.raku

my int $limit = 1_000_000;
my int8 @sieve[$limit + 1];
@sieve[$_] = 1 for 2..$limit;
loop (my int $i = 2; $i * $i <= $limit; $i++) {
    if @sieve[$i] {
        loop (my int $j = $i * $i; $j <= $limit; $j += $i) {
            @sieve[$j] = 0;
        }
    }
}
my int $count = 0;
loop ($i = 2; $i <= $limit; $i++) {
    $count++ if @sieve[$i];
}
say "Primes up to $limit: $count";
