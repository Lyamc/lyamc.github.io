(ns fibonacci)

(defn fib [n]
  (loop [i 0 a 0 b 1]
    (if (< i n)
      (recur (inc i) b (+ a b))
      a)))

(defn -main [& _]
  (println (fib 42)))

prime_sieve

Source: examples/clojure/prime_sieve.clj

(ns prime-sieve)

(defn sieve [limit]
  (let [is-prime (boolean-array (inc limit) true)]
    (aset is-prime 0 false)
    (aset is-prime 1 false)
    (doseq [i (range 2 (inc (int (Math/sqrt limit))))]
      (when (aget is-prime i)
        (loop [j (* i i)]
          (when (<= j limit)
            (aset is-prime j false)
            (recur (+ j i))))))
    (areduce is-prime i acc 0
             (if (aget is-prime i) (inc acc) acc))))

(defn -main [& _]
  (println (str "Primes up to 1000000: " (sieve 1000000))))

(ns sorting)

(defn -main [& _]
  (let [data (int-array 1000000)]
    (loop [i 0 seed (int 42)]
      (when (< i 1000000)
        (let [seed (unchecked-add (unchecked-multiply seed (int 1664525)) (int 1013904223))]
          (aset data i (bit-and seed (int 0x7fffffff)))
          (recur (inc i) seed))))
    (java.util.Arrays/sort data)
    (println (str "Sorted 1000000 integers: first=" (aget data 0)
                  ", last=" (aget data 999999)))))
