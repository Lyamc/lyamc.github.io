public class PrimeSieve {
    public static void main(String[] args) {
        final int LIMIT = 1_000_000;
        boolean[] sieve = new boolean[LIMIT + 1];
        for (int i = 2; i <= LIMIT; i++) sieve[i] = true;
        for (int i = 2; i * i <= LIMIT; i++) {
            if (sieve[i]) {
                for (int j = i * i; j <= LIMIT; j += i) {
                    sieve[j] = false;
                }
            }
        }
        int count = 0;
        for (int i = 2; i <= LIMIT; i++) {
            if (sieve[i]) count++;
        }
        System.out.println("Primes up to " + LIMIT + ": " + count);
    }
}

Source: examples/java/Sorting.java

public class Sorting {
    static void radixSort(int[] a) {
        int n = a.length;
        int[] tmp = new int[n];
        int[] src = a;
        int[] dst = tmp;
        int[] count = new int[256];
        for (int pass = 0; pass < 4; pass++) {
            java.util.Arrays.fill(count, 0);
            int shift = pass * 8;
            for (int i = 0; i < n; i++)
                count[(src[i] >>> shift) & 255]++;
            int sum = 0;
            for (int i = 0; i < 256; i++) {
                int c = count[i];
                count[i] = sum;
                sum += c;
            }
            for (int i = 0; i < n; i++) {
                int v = src[i];
                int b = (v >>> shift) & 255;
                dst[count[b]++] = v;
            }
            int[] t = src;
            src = dst;
            dst = t;
        }
    }

    public static void main(String[] args) {
        final int N = 1_000_000;
        int[] data = new int[N];
        int seed = 42;
        for (int i = 0; i < N; i++) {
            seed = seed * 1664525 + 1013904223;
            data[i] = seed & 0x7fffffff;
        }
        radixSort(data);
        System.out.println("Sorted " + N + " integers: first=" + data[0] + ", last=" + data[N - 1]);
    }
}
