prime_sieve

Source: examples/octave/prime_sieve.m

limit = 1000000;
sieve = true(1, limit + 1);
sieve(1) = false;
sieve(2) = false;
i = 2;
while i * i <= limit
  if sieve(i + 1)
    sieve((i*i+1):i:(limit+1)) = false;
  end
  i = i + 1;
end
count = sum(sieve(3:end));
printf("Primes up to %d: %d\n", limit, count);

sorting

Source: examples/octave/sorting.m

% LCG wrap is mod 2^32. Seed stays a double (product < 2^53, exact);
% results go into a preallocated uint32 vector (no per-iter uint32<->double).
% first/last match other uint32-wrap LCGs, not Octave's saturating uint32 arithmetic.
n = 1000000;
a = 1664525;
c = 1013904223;
two32 = 4294967296;
seed = 42;
data = zeros(1, n, "uint32");
for i = 1:n
  seed = mod(seed * a + c, two32);
  data(i) = seed;
end
data = bitand(data, uint32(2147483647));
data = sort(data);
printf("Sorted %d integers: first=%d, last=%d\n", n, data(1), data(n));
