prime_sieve

Source: examples/python/prime_sieve.py

def main():
    LIMIT = 1_000_000
    sieve = bytearray(b'\x01') * (LIMIT + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, int(LIMIT**0.5) + 1):
        if sieve[i]:
            sieve[i*i : LIMIT + 1 : i] = b'\x00' * len(sieve[i*i : LIMIT + 1 : i])
    count = sum(sieve)
    print(f"Primes up to {LIMIT}: {count}")

if __name__ == "__main__":
    main()

def main():
    n = 1_000_000
    data = [0] * n
    seed = 42
    for i in range(n):
        seed = (seed * 1664525 + 1013904223) & 0xFFFFFFFF
        data[i] = seed & 0x7FFFFFFF
    data.sort()
    print(f"Sorted {n} integers: first={data[0]}, last={data[-1]}")

if __name__ == "__main__":
    main()
