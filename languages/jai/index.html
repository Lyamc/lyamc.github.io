prime_sieve

Source: examples/jai/prime_sieve.jai

#import "Basic";

main :: () {
    LIMIT :: 1_000_000;
    sieve := NewArray(LIMIT + 1, u8);
    defer array_free(sieve);
    for i: 0..LIMIT {
        sieve[i] = 1;
    }
    sieve[0] = 0;
    sieve[1] = 0;
    i := 2;
    while i * i <= LIMIT {
        if sieve[i] != 0 {
            j := i * i;
            while j <= LIMIT {
                sieve[j] = 0;
                j += i;
            }
        }
        i += 1;
    }
    count := 0;
    for k: 2..LIMIT {
        if sieve[k] != 0  count += 1;
    }
    print("Primes up to %: %\n", LIMIT, count);
}

Source: examples/jai/sorting.jai

#import "Basic";

radix_sort :: (a: [] u32, tmp: [] u32, n: int) {
    src := a;
    dst := tmp;
    for pass: 0..3 {
        counts: [256] int;
        shift := pass * 8;
        for i: 0..n-1 {
            b := cast(int) ((src[i] >> shift) & 255);
            counts[b] += 1;
        }
        sum := 0;
        for b: 0..255 {
            c := counts[b];
            counts[b] = sum;
            sum += c;
        }
        for i: 0..n-1 {
            v := src[i];
            b := cast(int) ((v >> shift) & 255);
            dst[counts[b]] = v;
            counts[b] += 1;
        }
        t := src;
        src = dst;
        dst = t;
    }
}

main :: () {
    N :: 1_000_000;
    data := NewArray(N, u32);
    tmp := NewArray(N, u32);
    defer array_free(data);
    defer array_free(tmp);
    seed: u32 = 42;
    for i: 0..N-1 {
        seed = seed * 1664525 + 1013904223;
        data[i] = seed & 0x7fff_ffff;
    }
    radix_sort(data, tmp, N);
    print("Sorted % integers: first=%, last=%\n", N, data[0], data[N - 1]);
}
